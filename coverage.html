// Load loads configuration from file with Viper
func Load(configPath string) (*Config, error) {
        // Reset Viper to ensure clean state
        viper.Reset()

        // Set defaults first
        setDefaults()

        // Configure Viper
        viper.SetConfigType("yaml")
        viper.SetConfigName("config")

        if configPath != "" {
                viper.SetConfigFile(configPath)
        } else {
                // Add config search paths
                viper.AddConfigPath("./config")
                viper.AddConfigPath(".")
                viper.AddConfigPath("/etc/collabops")
        }

        // Enable environment variable support
        viper.SetEnvPrefix("COLLABOPS")
        viper.SetEnvKeyReplacer(strings.NewReplacer(".", "_"))
        viper.AutomaticEnv()

        // Read configuration file
        if err := viper.ReadInConfig(); err != nil {
                if _, ok := err.(viper.ConfigFileNotFoundError); ok {
                        // Config file not found; use defaults
                        // Note: Can't use logger here as it's not initialized yet
                } else if strings.Contains(err.Error(), "no such file or directory") {
                        // Specific file path doesn't exist; use defaults
                        // Note: Can't use logger here as it's not initialized yet
                } else {
                        return nil, fmt.Errorf("failed to read config file: %w", err)
                }
        }

        // Unmarshal configuration
        var config Config
        if err := viper.Unmarshal(&config); err != nil {
                return nil, fmt.Errorf("failed to unmarshal config: %w", err)
        }

        // Validate configuration
        if err := validate(&config); err != nil {

        return &config, nil
}

// validate performs configuration validation
func validate(config *Config) error {
        var errors []string

        // Validate server configuration
        if config.Server.Port < 1 || config.Server.Port > 65535 {
                errors = append(errors, "server.port must be between 1 and 65535")
        }

        if config.Server.BasePath == "" {
                errors = append(errors, "server.base_path is required")
        }

        if !strings.HasPrefix(config.Server.BasePath, "/") {
                errors = append(errors, "server.base_path must start with '/'")
        }

        // Validate Git configuration
        if config.Git.RepoRoot == "" {
                errors = append(errors, "git.repo_root is required")
        }

        if config.Git.MaxCacheSize < 0 {
                errors = append(errors, "git.max_cache_size must be >= 0")
        }

        // Validate JWT configuration
        validAlgorithms := []string{"RS256", "HS256", "ES256"}
        algorithmValid := false
        for _, alg := range validAlgorithms {
                if config.JWT.Algorithm == alg {
                        algorithmValid = true
                        break
                }
        }
        if !algorithmValid {
                errors = append(errors, fmt.Sprintf("jwt.algorithm must be one of: %v", validAlgorithms))
        }

        // For RS256 and ES256, public key path is required
        if (config.JWT.Algorithm == "RS256" || config.JWT.Algorithm == "ES256") && config.JWT.PublicKeyPath == "" {
                errors = append(errors, fmt.Sprintf("jwt.public_key_path is required for algorithm %s", config.JWT.Algorithm))
        }

        // For HS256, secret key is required
        if config.JWT.Algorithm == "HS256" && config.JWT.SecretKey == "" {
                errors = append(errors, "jwt.secret_key is required for algorithm HS256")
        }

        // Validate logging configuration
        validLogLevels := []string{"debug", "info", "warn", "error", "fatal"}
        logLevelValid := false
        for _, level := range validLogLevels {
                if config.Logging.Level == level {
                        logLevelValid = true
                        break
                }
        }
        if !logLevelValid {
                errors = append(errors, fmt.Sprintf("logging.level must be one of: %v", validLogLevels))
        }

        validLogFormats := []string{"json", "text"}
        logFormatValid := false
        for _, format := range validLogFormats {
                if config.Logging.Format == format {
                        logFormatValid = true
                        break
                }
        }
        if !logFormatValid {
                errors = append(errors, fmt.Sprintf("logging.format must be one of: %v", validLogFormats))
        }

        if len(errors) > 0 {
                return fmt.Errorf("validation errors: %s", strings.Join(errors, "; "))
        }

        return nil
}

// Reload reloads the configuration from file (thread-safe)
func (c *Config) Reload() error {
        c.mu.Lock()
        defer c.mu.Unlock()

        // Re-read configuration
        if err := viper.ReadInConfig(); err != nil {
                return fmt.Errorf("failed to reload config: %w", err)
        }

        // Create temporary config to validate
        var newConfig Config
        if err := viper.Unmarshal(&newConfig); err != nil {
                return fmt.Errorf("failed to unmarshal reloaded config: %w", err)
        }

        // Validate new configuration
        if err := validate(&newConfig); err != nil {
                return fmt.Errorf("reloaded config validation failed: %w", err)

        "container/list"
        "sync"
        "time"

// RepositoryCache manages cached Repository instances with LRU eviction
type RepositoryCache struct {
        mu      sync.RWMutex
        items   map[string]*cacheItem
        lruList *list.List
        maxSize int
        maxAge  time.Duration
        onEvict func(path string, repo Repository)
}

// cacheItem represents a cached repository with metadata
type cacheItem struct {
        path      string
        repo      Repository
        createdAt time.Time
        lastUsed  time.Time
        element   *list.Element
}

// CacheConfig holds configuration for the repository cache
type CacheConfig struct {
        // MaxSize is the maximum number of repositories to cache (0 for unlimited)
        MaxSize int

        // MaxAge is the maximum age of cached repositories (0 for no expiration)
        MaxAge time.Duration

        // OnEvict is called when a repository is evicted from the cache
        OnEvict func(path string, repo Repository)
}

// DefaultCacheConfig returns a default cache configuration
func DefaultCacheConfig() *CacheConfig {
        return &CacheConfig{
                MaxSize: 100,              // Cache up to 100 repositories
                MaxAge:  30 * time.Minute, // Expire after 30 minutes
                OnEvict: func(path string, repo Repository) {
                        // Default eviction handler closes the repository
                        if repo != nil {
                                _ = repo.Close()
                        }
                },
        }
}

// NewRepositoryCache creates a new repository cache with the given configuration
func NewRepositoryCache(config *CacheConfig) *RepositoryCache {
        if config == nil {
                config = DefaultCacheConfig()
        }

        return &RepositoryCache{
                items:   make(map[string]*cacheItem),
                lruList: list.New(),
                maxSize: config.MaxSize,
                maxAge:  config.MaxAge,
                onEvict: config.OnEvict,
        }
}

// Get retrieves a repository from the cache
func (c *RepositoryCache) Get(path string) (Repository, bool) {
        c.mu.Lock()
        defer c.mu.Unlock()

        item, exists := c.items[path]
        if !exists {
                return nil, false
        }

        // Check if item has expired
        if c.maxAge > 0 && time.Since(item.createdAt) > c.maxAge {
                c.removeItem(item)
                return nil, false
        }

        // Update last used time and move to front of LRU list
        item.lastUsed = time.Now()
        c.lruList.MoveToFront(item.element)

        return item.repo, true
}

// Put stores a repository in the cache
func (c *RepositoryCache) Put(path string, repo Repository) {
        c.mu.Lock()
        defer c.mu.Unlock()

        // If item already exists, update it
        if existing, exists := c.items[path]; exists {
                existing.repo = repo
                existing.lastUsed = time.Now()
                existing.createdAt = time.Now()
                c.lruList.MoveToFront(existing.element)
                return
        }

        // Create new cache item
        now := time.Now()
        item := &cacheItem{
                path:      path,
                repo:      repo,
                createdAt: now,
                lastUsed:  now,
        }

        // Add to front of LRU list
        item.element = c.lruList.PushFront(item)
        c.items[path] = item

        // Evict oldest items if cache is full
        c.evictIfNeeded()
}

// Remove removes a repository from the cache
func (c *RepositoryCache) Remove(path string) bool {
        c.mu.Lock()
        defer c.mu.Unlock()

        item, exists := c.items[path]
        if !exists {
                return false
        }

        c.removeItem(item)
        return true
}

// Clear removes all repositories from the cache
func (c *RepositoryCache) Clear() {
        c.mu.Lock()
        defer c.mu.Unlock()

        // Call eviction callback for all items
        for _, item := range c.items {
                if c.onEvict != nil {
                        c.onEvict(item.path, item.repo)
                }
        }

        // Clear all data structures
        c.items = make(map[string]*cacheItem)
        c.lruList.Init()
}

// Size returns the current number of cached repositories
func (c *RepositoryCache) Size() int {
        c.mu.RLock()
        defer c.mu.RUnlock()
        return len(c.items)
}

// Stats returns cache statistics
func (c *RepositoryCache) Stats() CacheStats {
        c.mu.RLock()
        defer c.mu.RUnlock()

        stats := CacheStats{
                Size:    len(c.items),
                MaxSize: c.maxSize,
                MaxAge:  c.maxAge,
        }

        // Calculate oldest and newest items
        if len(c.items) > 0 {
                now := time.Now()
                oldest := now
                newest := time.Time{}

                for _, item := range c.items {
                        if item.createdAt.Before(oldest) {
                                oldest = item.createdAt
                        }
                        if item.createdAt.After(newest) {
                                newest = item.createdAt
                        }
                }

                stats.OldestAge = now.Sub(oldest)
                stats.NewestAge = now.Sub(newest)
        }

        return stats
}

// CleanupExpired removes all expired items from the cache
func (c *RepositoryCache) CleanupExpired() int {
        if c.maxAge <= 0 {
                return 0 // No expiration configured
        }

        c.mu.Lock()
        defer c.mu.Unlock()

        var expiredItems []*cacheItem
        cutoff := time.Now().Add(-c.maxAge)

        // Find expired items
        for _, item := range c.items {
                if item.createdAt.Before(cutoff) {
                        expiredItems = append(expiredItems, item)
                }
        }

        // Remove expired items
        for _, item := range expiredItems {
                c.removeItem(item)
        }

        return len(expiredItems)
}

// removeItem removes an item from the cache (must be called with lock held)
func (c *RepositoryCache) removeItem(item *cacheItem) {
        // Remove from map
        delete(c.items, item.path)

        // Remove from LRU list
        c.lruList.Remove(item.element)

        // Call eviction callback
        if c.onEvict != nil {
                c.onEvict(item.path, item.repo)
        }
}

// evictIfNeeded evicts oldest items if cache size exceeds maximum (must be called with lock held)
func (c *RepositoryCache) evictIfNeeded() {
        if c.maxSize <= 0 {
                return // No size limit configured
        }

        // Evict oldest items until we're under the size limit
        for len(c.items) > c.maxSize {
                // Get oldest item from back of LRU list
                oldest := c.lruList.Back()
                if oldest == nil {
                        break
                }

                item := oldest.Value.(*cacheItem)
                c.removeItem(item)
        }
}

// CacheStats provides statistics about the cache
type CacheStats struct {
        Size      int           `json:"size"`
        MaxSize   int           `json:"max_size"`
        MaxAge    time.Duration `json:"max_age"`
        OldestAge time.Duration `json:"oldest_age"`
        NewestAge time.Duration `json:"newest_age"`
}

// Global cache instance (can be overridden for testing)
var globalCache = NewRepositoryCache(DefaultCacheConfig())

// SetGlobalCache sets the global repository cache instance
func SetGlobalCache(cache *RepositoryCache) {
        globalCache = cache
}

// GetGlobalCache returns the global repository cache instance
func GetGlobalCache() *RepositoryCache {
        return globalCache
}

// ClearGlobalCache clears the global repository cache
func ClearGlobalCache() {
        globalCache.Clear()
}

// GetGlobalCacheStats returns statistics for the global cache
func GetGlobalCacheStats() CacheStats {
        return globalCache.Stats()
}

// GoGitRepository implements the Repository interface using go-git
type GoGitRepository struct {
        path string
        repo *git.Repository
}

// OpenRepository opens a Git repository at the specified path
func OpenRepository(path string) (Repository, error) {
        return openRepositoryInternal(path, false)
}

// OpenRepositoryWithCache opens a Git repository at the specified path using the global cache
func OpenRepositoryWithCache(path string) (Repository, error) {
        return openRepositoryInternal(path, true)
}

// openRepositoryInternal is the internal implementation that optionally uses caching
func openRepositoryInternal(path string, useCache bool) (Repository, error) {
        // If caching is enabled, try to get from cache first
        if useCache {
                if cached, found := globalCache.Get(path); found {
                        return cached, nil
                }
        }

        // Check if the path exists
        if _, err := os.Stat(path); os.IsNotExist(err) {
                return nil, &RepositoryNotFoundError{Path: path}
        }

        // Try to open the repository
        repo, err := git.PlainOpen(path)
        if err != nil {
                // Check if it's because the directory is not a git repository
                if err == git.ErrRepositoryNotExists {
                        return nil, &InvalidRepositoryError{
                                Path:   path,
                                Reason: "directory is not a Git repository",
                        }
                }

                return nil, NewGitError("open", path, "", "failed to open repository", err)
        }

        // Create our repository wrapper
        gitRepo := &GoGitRepository{
                path: path,
                repo: repo,
        }

        // If caching is enabled, store in cache
        if useCache {
                globalCache.Put(path, gitRepo)
        }

        return gitRepo, nil
}

// Path returns the filesystem path of the repository
func (r *GoGitRepository) Path() string {
        return r.path
}

// GetTree retrieves the directory structure at a given reference
func (r *GoGitRepository) GetTree(ctx context.Context, ref string) ([]TreeEntry, error) {
        // Check for context cancellation
        select {
        case <-ctx.Done():
                return nil, ctx.Err()
        default:
        }

        // Resolve the reference to a commit
        commit, err := r.resolveCommit(ref)
        if err != nil {
                return nil, err
        }

        // Get the tree from the commit
        tree, err := commit.Tree()
        if err != nil {
                return nil, NewGitError("get_tree", r.path, ref, "failed to get tree from commit", err)
        }

        // Convert tree entries to our format
        var entries []TreeEntry
        for _, entry := range tree.Entries {
                // Check for context cancellation during iteration
                select {
                case <-ctx.Done():
                        return nil, ctx.Err()
                default:
                }

                entryType := convertFileMode(entry.Mode)
                size := int64(0)

                // For blobs, try to get the size
                if entryType == EntryTypeBlob {
                        if blob, err := r.repo.BlobObject(entry.Hash); err == nil {
                                size = blob.Size
                        }
                }

                entries = append(entries, TreeEntry{
                        Path: entry.Name,
                        Mode: entry.Mode.String(),
                        Type: entryType,
                        Hash: entry.Hash.String(),
                        Size: size,
                })
        }

        return entries, nil
}

// GetBlob retrieves the contents of a file at a specific path and reference
func (r *GoGitRepository) GetBlob(ctx context.Context, path, ref string) ([]byte, error) {
        // Check for context cancellation
        select {
        case <-ctx.Done():
                return nil, ctx.Err()
        default:
        }

        // Resolve the reference to a commit
        commit, err := r.resolveCommit(ref)
        if err != nil {
                return nil, err
        }

        // Get the file from the commit
        file, err := commit.File(path)
        if err != nil {
                if err == object.ErrFileNotFound {
                        return nil, &PathNotFoundError{Path: path, Ref: ref, Repo: r.path}
                }
                return nil, NewGitError("get_blob", r.path, ref, fmt.Sprintf("failed to get file %s", path), err)
        }

        // Read the file contents
        reader, err := file.Reader()
        if err != nil {
                return nil, NewGitError("get_blob", r.path, ref, fmt.Sprintf("failed to get reader for file %s", path), err)
        }
        defer func() { _ = reader.Close() }()

        // Check for context cancellation before reading
        select {
        case <-ctx.Done():
                return nil, ctx.Err()
        default:
        }

        // Read all content
        content, err := io.ReadAll(reader)
        if err != nil {
                return nil, NewGitError("get_blob", r.path, ref, fmt.Sprintf("failed to read file %s", path), err)
        }

        return content, nil
}

// GetCommits retrieves commit history with pagination
func (r *GoGitRepository) GetCommits(ctx context.Context, ref string, limit, skip int) ([]Commit, error) {
        // Check for context cancellation
        select {
        case <-ctx.Done():
                return nil, ctx.Err()
        default:
        }

        // Resolve the reference to a commit hash
        hash, err := r.resolveReference(ref)
        if err != nil {
                return nil, err
        }

        // Get commit iterator starting from the resolved hash
        commitIter, err := r.repo.Log(&git.LogOptions{
                From: hash,
        })
        if err != nil {
                return nil, NewGitError("get_commits", r.path, ref, "failed to get commit log", err)
        }
        defer commitIter.Close()

        var commits []Commit
        current := 0

        err = commitIter.ForEach(func(c *object.Commit) error {
                // Check for context cancellation during iteration
                select {
                case <-ctx.Done():
                        return ctx.Err()
                default:
                }

                // Skip commits if needed
                if current < skip {
                        current++
                        return nil
                }

                // Stop if we've reached our limit
                if len(commits) >= limit {
                        return io.EOF // Signal to stop iteration
                }

                // Convert to our commit format
                commit := convertCommit(c)
                commits = append(commits, commit)
                current++

                return nil
        })

        if err != nil && err != io.EOF {
                return nil, NewGitError("get_commits", r.path, ref, "failed to iterate commits", err)
        }

        return commits, nil
}

// Close releases any resources held by the repository
func (r *GoGitRepository) Close() error {
        // go-git repositories don't need explicit closing
        // but we can set the repo to nil to help with garbage collection
        r.repo = nil
        return nil
}

// Helper methods

// resolveCommit resolves a reference string to a commit object
func (r *GoGitRepository) resolveCommit(ref string) (*object.Commit, error) {
        hash, err := r.resolveReference(ref)
        if err != nil {
                return nil, err
        }

        commit, err := r.repo.CommitObject(hash)
        if err != nil {
                return nil, NewGitError("resolve_commit", r.path, ref, "failed to get commit object", err)
        }

        return commit, nil
}

// resolveReference resolves a reference string to a hash
func (r *GoGitRepository) resolveReference(ref string) (plumbing.Hash, error) {
        // Try to parse as a direct hash first
        if hash := plumbing.NewHash(ref); !hash.IsZero() && len(ref) >= 4 {
                // Verify the object exists
                if _, err := r.repo.Object(plumbing.AnyObject, hash); err == nil {
                        return hash, nil
                }
        }

        // Try to resolve as a reference (branch, tag, etc.)
        reference, err := r.repo.Reference(plumbing.ReferenceName(ref), true)
        if err != nil {
                // Try with refs/heads/ prefix for branches
                if branchRef, branchErr := r.repo.Reference(plumbing.NewBranchReferenceName(ref), true); branchErr == nil {
                        return branchRef.Hash(), nil
                }

                // Try with refs/tags/ prefix for tags
                if tagRef, tagErr := r.repo.Reference(plumbing.NewTagReferenceName(ref), true); tagErr == nil {
                        return tagRef.Hash(), nil
                }

                // Try HEAD if ref is empty or "HEAD"
                if ref == "" || ref == "HEAD" {
                        if headRef, headErr := r.repo.Head(); headErr == nil {
                                return headRef.Hash(), nil
                        }
                }

                return plumbing.ZeroHash, &ReferenceNotFoundError{Ref: ref, Path: r.path}
        }

        return reference.Hash(), nil
}

// convertFileMode converts go-git file mode to our EntryType
func convertFileMode(mode filemode.FileMode) EntryType {
        switch mode {
        case filemode.Dir: // Directory (040000)
                return EntryTypeTree
        case filemode.Submodule: // Git submodule (160000)
                return EntryTypeCommit
        default: // Regular file, executable, symlink
                return EntryTypeBlob
        }
}

// convertCommit converts a go-git commit to our Commit format
func convertCommit(c *object.Commit) Commit {
        var parentHashes []string
        for _, parent := range c.ParentHashes {
                parentHashes = append(parentHashes, parent.String())
        }

        return Commit{
                Hash:    c.Hash.String(),
                Message: c.Message,
                Author: CommitAuthor{
                        Name:  c.Author.Name,
                        Email: c.Author.Email,
                        Date:  c.Author.When,
                },
                Committer: CommitAuthor{
                        Name:  c.Committer.Name,
                        Email: c.Committer.Email,
                        Date:  c.Committer.When,
                },
                ParentHashes: parentHashes,
                TreeHash:     c.TreeHash.String(),
        }
}

        }

        return &AppError{
                Type:        b.errorType,
                Code:        b.code,
                Message:     b.message,
                Details:     b.details,
                Cause:       b.cause,
                Context:     b.context,
                StatusCode:  statusCode,
                Retryable:   b.retryable,
                UserMessage: b.userMessage,
        }
}

// getDefaultStatusCode returns default HTTP status codes for error types
func getDefaultStatusCode(errorType ErrorType) int {
        switch errorType {
        case ErrorTypeValidation:
                return http.StatusBadRequest
        case ErrorTypeNotFound:
                return http.StatusNotFound
        case ErrorTypeUnauthorized:
                return http.StatusUnauthorized
        case ErrorTypeForbidden:
                return http.StatusForbidden
        case ErrorTypeConflict:
                return http.StatusConflict
        case ErrorTypeTimeout:
                return http.StatusRequestTimeout
        case ErrorTypeRateLimit:
                return http.StatusTooManyRequests
        case ErrorTypeExternal:
                return http.StatusBadGateway
        default:
                return http.StatusInternalServerError
        }
}

// Handle processes an error and returns appropriate HTTP response
func (h *ErrorHandler) Handle(ctx context.Context, w http.ResponseWriter, err error) {
        // Extract request ID from context
        requestID := ""
        if id, ok := ctx.Value(RequestIDKey).(string); ok {
                requestID = id
        }

        var appErr *AppError

        // Convert to AppError if it's not already
        if ae, ok := err.(*AppError); ok {
                appErr = ae
                if appErr.RequestID == "" {
                        appErr.RequestID = requestID
                }
        } else {
                // Wrap unknown errors as internal errors
                appErr = NewInternalError("UNKNOWN_ERROR", "An unexpected error occurred").
                        WithCause(err).
                        WithRequestID(requestID)
        }

        // Log the error
        h.logError(ctx, appErr)

        // Send HTTP response
        h.sendErrorResponse(w, appErr)
}

// logError logs the error with appropriate level and context
func (h *ErrorHandler) logError(ctx context.Context, appErr *AppError) {
        fields := []zap.Field{
                zap.String("error_type", string(appErr.Type)),
                zap.String("error_code", appErr.Code),
                zap.String("error_message", appErr.Message),
                zap.Bool("retryable", appErr.Retryable),
                zap.Int("status_code", appErr.StatusCode),
        }

        if appErr.RequestID != "" {
                fields = append(fields, zap.String("request_id", appErr.RequestID))
        }

        if appErr.Details != "" {
                fields = append(fields, zap.String("details", appErr.Details))
        }

        // Add context fields
        for key, value := range appErr.Context {
                fields = append(fields, zap.Any(fmt.Sprintf("ctx_%s", key), value))
        }

        // Log at appropriate level
        switch appErr.Type {
        case ErrorTypeValidation, ErrorTypeNotFound, ErrorTypeUnauthorized, ErrorTypeForbidden:
                h.logger.LogWarning(ctx, "Client error", fields...)
        case ErrorTypeInternal, ErrorTypeExternal:
                h.logger.LogError(ctx, appErr.Cause, "Server error", fields...)
        default:
                h.logger.LogError(ctx, appErr.Cause, "Application error", fields...)
        }
}

// sendErrorResponse sends an HTTP error response
func (h *ErrorHandler) sendErrorResponse(w http.ResponseWriter, appErr *AppError) {
        w.Header().Set("Content-Type", "application/json")
        w.WriteHeader(appErr.StatusCode)

        // Create response body
        response := map[string]interface{}{
                "error": map[string]interface{}{
                        "type":    appErr.Type,
                        "code":    appErr.Code,
                        "message": appErr.UserMessage,
                },
        }

        if appErr.RequestID != "" {
                response["request_id"] = appErr.RequestID
        }

        // Include details in non-production environments or for client errors
        if appErr.Type != ErrorTypeInternal || appErr.Details != "" {
                response["error"].(map[string]interface{})["details"] = appErr.Details
        }

        // Encode and send response
        if err := json.NewEncoder(w).Encode(response); err != nil {
                // Fallback to plain text if JSON encoding fails
                http.Error(w, "Internal Server Error", http.StatusInternalServerError)
        }
}

// HandlePanic recovers from panics and converts them to errors
func (h *ErrorHandler) HandlePanic(ctx context.Context, w http.ResponseWriter) {
        if r := recover(); r != nil {
                var err error
                if e, ok := r.(error); ok {
                        err = e
                } else {
                        err = fmt.Errorf("panic: %v", r)
                }

                appErr := NewInternalError("PANIC", "Request processing failed").
                        WithCause(err).
                        WithDetails(fmt.Sprintf("Panic recovered: %v", r))

                h.Handle(ctx, w, appErr)
        }
}

// NewLogger creates a new logger instance with the given configuration
func NewLogger(cfg config.LoggingConfig) (*Logger, error) {
        // Create encoder config
        encoderConfig := zapcore.EncoderConfig{
                TimeKey:        "timestamp",
                LevelKey:       "level",
                NameKey:        "logger",
                CallerKey:      "caller",
                FunctionKey:    "function",
                MessageKey:     "message",
                StacktraceKey:  "stacktrace",
                LineEnding:     zapcore.DefaultLineEnding,
                EncodeLevel:    zapcore.LowercaseLevelEncoder,
                EncodeTime:     zapcore.ISO8601TimeEncoder,
                EncodeDuration: zapcore.StringDurationEncoder,
                EncodeCaller:   zapcore.ShortCallerEncoder,
        }

        // Create encoder based on format
        var encoder zapcore.Encoder
        switch cfg.Format {
        case "json":
                encoder = zapcore.NewJSONEncoder(encoderConfig)
        case "text":
                encoderConfig.EncodeLevel = zapcore.CapitalColorLevelEncoder
                encoderConfig.EncodeTime = zapcore.RFC3339TimeEncoder
                encoder = zapcore.NewConsoleEncoder(encoderConfig)
        default:
                return nil, fmt.Errorf("unsupported log format: %s", cfg.Format)
        }

        // Parse log level
        level, err := zapcore.ParseLevel(cfg.Level)
        if err != nil {
                return nil, fmt.Errorf("invalid log level %s: %w", cfg.Level, err)
        }

        // Create output writers
        var writeSyncer zapcore.WriteSyncer
        switch cfg.Output {
        case "stdout":
                writeSyncer = zapcore.AddSync(os.Stdout)
        case "stderr":
                writeSyncer = zapcore.AddSync(os.Stderr)
        default:
                // File output with rotation
                logRotator := &lumberjack.Logger{
                        Filename:   cfg.Output,
                        MaxSize:    100, // 100 MB
                        MaxBackups: 5,
                        MaxAge:     30, // 30 days
                        Compress:   true,
                }
                writeSyncer = zapcore.AddSync(logRotator)
        }

        // Create core
        core := zapcore.NewCore(encoder, writeSyncer, level)

        // Create logger with additional options
        options := []zap.Option{
                zap.AddCaller(),
                zap.AddStacktrace(zapcore.ErrorLevel),
        }

        if cfg.Development {
                options = append(options, zap.Development())
        }

        zapLogger := zap.New(core, options...)

        return &Logger{
                Logger: zapLogger,
        }, nil
}

// RequestIDMiddleware generates and adds request IDs to the context
func RequestIDMiddleware(generator RequestIDGenerator) func(http.Handler) http.Handler {
        if generator == nil {
                generator = &DefaultRequestIDGenerator{}
        }

        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        requestID := generator.Generate()

                        // Add request ID to context
                        ctx := context.WithValue(r.Context(), RequestIDKey, requestID)

                        // Add request ID to response header for tracing
                        w.Header().Set("X-Request-ID", requestID)

                        // Continue with the request
                        next.ServeHTTP(w, r.WithContext(ctx))
                })
        }
}

// CorrelationIDMiddleware extracts correlation ID from headers and adds to context
func CorrelationIDMiddleware() func(http.Handler) http.Handler {
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        // Extract correlation ID from header
                        correlationID := r.Header.Get("X-Correlation-ID")
                        if correlationID == "" {
                                // If not provided, use request ID as correlation ID
                                if requestID, ok := r.Context().Value(RequestIDKey).(string); ok {
                                        correlationID = requestID
                                }
                        }

                        // Add correlation ID to context
                        ctx := context.WithValue(r.Context(), CorrelationIDKey, correlationID)

                        // Add correlation ID to response header
                        w.Header().Set("X-Correlation-ID", correlationID)

                        // Continue with the request
                        next.ServeHTTP(w, r.WithContext(ctx))
                })
        }
}

// LoggingMiddleware logs HTTP requests with context
func LoggingMiddleware(logger *Logger) func(http.Handler) http.Handler {
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        start := time.Now()

                        // Wrap response writer to capture status code
                        ww := middleware.NewWrapResponseWriter(w, r.ProtoMajor)

                        // Continue with the request
                        next.ServeHTTP(ww, r)

                        // Log the request
                        duration := time.Since(start)
                        logger.HTTPRequest(
                                r.Context(),
                                r.Method,
                                r.URL.Path,
                                r.UserAgent(),
                                r.RemoteAddr,
                                ww.Status(),
                                duration,
                        )
                })
        }
}

// ErrorHandlingMiddleware provides centralized error handling and logging
func ErrorHandlingMiddleware(logger *Logger) func(http.Handler) http.Handler {
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        defer func() {
                                if err := recover(); err != nil {
                                        // Log the panic
                                        logger.LogError(
                                                r.Context(),
                                                err.(error),
                                                "HTTP request panic",
                                                zap.String("method", r.Method),
                                                zap.String("path", r.URL.Path),
                                                zap.String("user_agent", r.UserAgent()),
                                                zap.String("remote_addr", r.RemoteAddr),
                                        )

                                        // Return 500 Internal Server Error
                                        http.Error(w, "Internal Server Error", http.StatusInternalServerError)
                                }
                        }()

                        next.ServeHTTP(w, r)
                })
        }
}

// HealthCheckMiddleware skips logging for health check endpoints
func HealthCheckMiddleware(healthPaths ...string) func(http.Handler) http.Handler {
        healthMap := make(map[string]bool)
        for _, path := range healthPaths {
                healthMap[path] = true
        }
        // Default health check paths
        healthMap["/health"] = true
        healthMap["/healthz"] = true
        healthMap["/ping"] = true

        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        // Skip logging for health check endpoints
                        if healthMap[r.URL.Path] {
                                // Add a marker to context to skip logging
                                ctx := context.WithValue(r.Context(), SkipLoggingKey, true)
                                next.ServeHTTP(w, r.WithContext(ctx))
                        } else {
                                next.ServeHTTP(w, r)
                        }
                })
        }
}

// ConditionalLoggingMiddleware only logs if the context doesn't have skip_logging
func ConditionalLoggingMiddleware(logger *Logger) func(http.Handler) http.Handler {
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        start := time.Now()

                        // Wrap response writer to capture status code
                        ww := middleware.NewWrapResponseWriter(w, r.ProtoMajor)

                        // Continue with the request
                        next.ServeHTTP(ww, r)

                        // Only log if not marked to skip
                        if skip, ok := r.Context().Value(SkipLoggingKey).(bool); !ok || !skip {
                                duration := time.Since(start)
                                logger.HTTPRequest(
                                        r.Context(),
                                        r.Method,
                                        r.URL.Path,
                                        r.UserAgent(),
                                        r.RemoteAddr,
                                        ww.Status(),
                                        duration,
                                )
                        }
                })
        }
}

// GetLoggerFromContext extracts logger from request context
func GetLoggerFromContext(ctx context.Context) *zap.Logger {
        logger := GetGlobalLogger()
        return logger.WithContext(ctx)
}

// LogMiddleware sets up the complete logging middleware stack
func LogMiddleware(logger *Logger) []func(http.Handler) http.Handler {
        return []func(http.Handler) http.Handler{
                RequestIDMiddleware(nil), // Use default generator
                CorrelationIDMiddleware(),
                ErrorHandlingMiddleware(logger),
                HealthCheckMiddleware(), // Skip logging for health checks
                ConditionalLoggingMiddleware(logger),
        }
}

// SetupLoggingMiddleware is a convenience function to set up all logging middleware
func SetupLoggingMiddleware(r chi.Router, logger *Logger) {
        middlewares := LogMiddleware(logger)
        for _, mw := range middlewares {
                r.Use(mw)
        }
}

// PermissionCache provides thread-safe caching of permission check results.
// It uses sync.Map for concurrent access and includes TTL-based expiration.

        ttl   time.Duration
        // We could add a cleanup goroutine here in the future for automatic expiration

// NewPermissionCache creates a new permission cache with the specified TTL.

                cache: sync.Map{},
                ttl:   ttl,
        }
}

// Get retrieves a permission result from the cache.
// Returns nil if the key doesn't exist.
func (pc *PermissionCache) Get(key string) *PermissionResult {
        value, exists := pc.cache.Load(key)
        if !exists {
                return nil
        }

        result, ok := value.(*PermissionResult)
        if !ok {
                // This should never happen, but handle it gracefully
                pc.cache.Delete(key)
                return nil
        }

        return result
}

// Set stores a permission result in the cache.
func (pc *PermissionCache) Set(key string, result *PermissionResult) {
        pc.cache.Store(key, result)
}

// Delete removes a permission result from the cache.
func (pc *PermissionCache) Delete(key string) {
        pc.cache.Delete(key)
}

// Clear removes all entries from the cache.
func (pc *PermissionCache) Clear() {
        pc.cache.Range(func(key, value interface{}) bool {
                pc.cache.Delete(key)
                return true
        })
}

// Close gracefully shuts down the cache.
// Currently, this doesn't do anything since sync.Map doesn't require cleanup,
// but it's here for future extensibility (e.g., stopping cleanup goroutines).
func (pc *PermissionCache) Close() {
        // Future: Stop any background cleanup goroutines
}

// CleanupExpired removes expired entries from the cache.
// This is a manual cleanup method that can be called periodically.
func (pc *PermissionCache) CleanupExpired() {
        now := time.Now()
        pc.cache.Range(func(key, value interface{}) bool {
                result, ok := value.(*PermissionResult)
                if !ok {
                        // Invalid entry, remove it
                        pc.cache.Delete(key)
                        return true
                }

                if now.After(result.ExpiresAt) {
                        pc.cache.Delete(key)
                }
                return true
        })
}

// Size returns the approximate number of entries in the cache.
// Note: This iterates through all entries, so it's O(n) and should be used sparingly.
func (pc *PermissionCache) Size() int {
        count := 0
        pc.cache.Range(func(key, value interface{}) bool {
                count++
                return true
        })
        return count
}

                return "open"
        case StateHalfOpen:
                return "half-open"
        default:
                return "unknown"
        }
}

// CircuitBreakerConfig holds configuration for the circuit breaker.
type CircuitBreakerConfig struct {
        // FailureThreshold is the number of failures required to open the circuit.
        FailureThreshold int

        // RecoveryTimeout is the time to wait before attempting to close the circuit.
        RecoveryTimeout time.Duration

        // MaxHalfOpenRequests is the maximum number of requests allowed in half-open state.
        MaxHalfOpenRequests int
}

// DefaultCircuitBreakerConfig returns a configuration with sensible defaults.
func DefaultCircuitBreakerConfig() *CircuitBreakerConfig {
        return &CircuitBreakerConfig{
                FailureThreshold:    5,
                RecoveryTimeout:     30 * time.Second,
                MaxHalfOpenRequests: 1,
        }
}

// CircuitBreaker implements the circuit breaker pattern.
type CircuitBreaker struct {
        config          *CircuitBreakerConfig
        state           CircuitBreakerState
        failureCount    int
        halfOpenCount   int
        lastFailureTime time.Time
        lastSuccessTime time.Time
        mutex           sync.RWMutex
}

// NewCircuitBreaker creates a new circuit breaker with the given configuration.
func NewCircuitBreaker(config *CircuitBreakerConfig) *CircuitBreaker {
        if config == nil {
                config = DefaultCircuitBreakerConfig()
        }

        return &CircuitBreaker{
                config: config,
                state:  StateClosed,
        }
}

// Execute runs the given function if the circuit allows it.
func (cb *CircuitBreaker) Execute(ctx context.Context, fn func(context.Context) error) error {
        if !cb.canExecute() {
                return &CircuitBreakerError{
                        State:   cb.getState(),
                        Message: "circuit breaker is open",
                }
        }

        err := fn(ctx)
        cb.recordResult(err)
        return err
}

// canExecute determines if a request can be executed based on circuit state.
func (cb *CircuitBreaker) canExecute() bool {
        cb.mutex.RLock()
        defer cb.mutex.RUnlock()

        switch cb.state {
        case StateClosed:
                return true
        case StateOpen:
                // Check if recovery timeout has passed
                return time.Since(cb.lastFailureTime) >= cb.config.RecoveryTimeout
        case StateHalfOpen:
                // Allow limited requests in half-open state
                return cb.halfOpenCount < cb.config.MaxHalfOpenRequests
        default:
                return false
        }
}

// recordResult updates the circuit breaker state based on the operation result.
func (cb *CircuitBreaker) recordResult(err error) {
        cb.mutex.Lock()
        defer cb.mutex.Unlock()

        if err != nil {
                cb.onFailure()
        } else {
                cb.onSuccess()
        }
}

// onFailure handles a failed operation.
func (cb *CircuitBreaker) onFailure() {
        cb.failureCount++
        cb.lastFailureTime = time.Now()

        switch cb.state {
        case StateClosed:
                if cb.failureCount >= cb.config.FailureThreshold {
                        cb.setState(StateOpen)
                }
        case StateHalfOpen:
                cb.setState(StateOpen)
                cb.halfOpenCount = 0
        }
}

// onSuccess handles a successful operation.
func (cb *CircuitBreaker) onSuccess() {
        cb.lastSuccessTime = time.Now()

        switch cb.state {
        case StateClosed:
                cb.failureCount = 0
        case StateHalfOpen:
                cb.setState(StateClosed)
                cb.failureCount = 0
                cb.halfOpenCount = 0
        case StateOpen:
                // This shouldn't happen, but reset if it does
                cb.setState(StateClosed)
                cb.failureCount = 0
        }
}

// setState updates the circuit breaker state.
func (cb *CircuitBreaker) setState(newState CircuitBreakerState) {
        if cb.state != newState {
                cb.state = newState

                if newState == StateHalfOpen {
                        cb.halfOpenCount = 0
                }
        }
}

// getState returns the current state of the circuit breaker.
func (cb *CircuitBreaker) getState() CircuitBreakerState {
        cb.mutex.RLock()
        defer cb.mutex.RUnlock()

        // Transition from open to half-open if recovery timeout has passed
        if cb.state == StateOpen && time.Since(cb.lastFailureTime) >= cb.config.RecoveryTimeout {
                cb.mutex.RUnlock()
                cb.mutex.Lock()
                // Double-check after acquiring write lock
                if cb.state == StateOpen && time.Since(cb.lastFailureTime) >= cb.config.RecoveryTimeout {
                        cb.setState(StateHalfOpen)
                }
                cb.mutex.Unlock()
                cb.mutex.RLock()
        }

        return cb.state
}

// GetMetrics returns metrics about the circuit breaker.
func (cb *CircuitBreaker) GetMetrics() CircuitBreakerMetrics {
        cb.mutex.RLock()
        defer cb.mutex.RUnlock()

        return CircuitBreakerMetrics{
                State:             cb.state,
                FailureCount:      cb.failureCount,
                HalfOpenCount:     cb.halfOpenCount,
                LastFailureTime:   cb.lastFailureTime,
                LastSuccessTime:   cb.lastSuccessTime,
                TimeUntilHalfOpen: cb.timeUntilHalfOpen(),
        }
}

// timeUntilHalfOpen calculates time remaining until half-open state.
func (cb *CircuitBreaker) timeUntilHalfOpen() time.Duration {
        if cb.state != StateOpen {
                return 0
        }

        elapsed := time.Since(cb.lastFailureTime)
        if elapsed >= cb.config.RecoveryTimeout {
                return 0
        }

        return cb.config.RecoveryTimeout - elapsed
}

// CircuitBreakerMetrics holds metrics about the circuit breaker.
type CircuitBreakerMetrics struct {
        State             CircuitBreakerState
        FailureCount      int
        HalfOpenCount     int
        LastFailureTime   time.Time
        LastSuccessTime   time.Time
        TimeUntilHalfOpen time.Duration
}

// CircuitBreakerError represents an error when the circuit breaker is open.
type CircuitBreakerError struct {
        State   CircuitBreakerState
        Message string
}

func (e *CircuitBreakerError) Error() string {
        return e.Message
}

// FallbackPolicy defines the behavior when the API-Service is unavailable.
type FallbackPolicy int

const (
        // FallbackDeny denies all requests when the service is unavailable.
        FallbackDeny FallbackPolicy = iota
        // FallbackAllow allows all requests when the service is unavailable.
        FallbackAllow
        // FallbackReadOnly allows only read requests when the service is unavailable.
        FallbackReadOnly
        // FallbackUseCache uses cached results only, denying requests without cache entries.
        FallbackUseCache
)

func (p FallbackPolicy) String() string {
        switch p {
        case FallbackDeny:
                return "deny"
        case FallbackAllow:
                return "allow"
        case FallbackReadOnly:
                return "read-only"
        case FallbackUseCache:
                return "use-cache"
        default:
                return "unknown"
        }
}

// ShouldAllowAccess determines if access should be allowed based on the fallback policy.
func (p FallbackPolicy) ShouldAllowAccess(operation Operation, hasCachedResult bool) bool {
        switch p {
        case FallbackDeny:
                return false
        case FallbackAllow:
                return true
        case FallbackReadOnly:
                return operation == OperationRead
        case FallbackUseCache:
                return hasCachedResult
        default:
                return false
        }
}

        if config == nil {

        if err := config.Validate(); err != nil {
                return nil, errors.New("invalid configuration: " + err.Error())

                config:         config,
                httpClient:     httpClient,
                cache:          cache,
                circuitBreaker: circuitBreaker,

// checkAccess is the internal method that performs the actual permission check.
func (c *Client) checkAccess(ctx context.Context, userID, repoOwner, repoName string, operation Operation) (bool, error) {
        // Input validation
        if userID == "" {
                return false, errors.New("userID cannot be empty")
        }
        if repoOwner == "" {
                return false, errors.New("repoOwner cannot be empty")
        }
        if repoName == "" {
                return false, errors.New("repoName cannot be empty")
        }

        // Check cache first
        result := &PermissionResult{
                UserID:    userID,
                RepoOwner: repoOwner,
                RepoName:  repoName,
                Operation: operation,
        }

        cacheKey := result.CacheKey()
        cachedResult := c.cache.Get(cacheKey)

        // Return cached result if valid
        if cachedResult != nil && !cachedResult.IsExpired() {
                return cachedResult.HasAccess, nil
        }

        // Attempt API request with circuit breaker protection
        hasAccess, err := c.requestPermissionWithFallback(ctx, userID, repoOwner, repoName, operation, cachedResult)
        if err != nil {
                // Check if this is a ServiceUnavailableError with fallback information
                var svcErr *ServiceUnavailableError
                if errors.As(err, &svcErr) && svcErr.FallbackUsed {
                        // Return the fallback access decision with the error
                        return hasAccess, err
                }
                return false, err
        }

        // Cache the successful result
        result.HasAccess = hasAccess
        result.CachedAt = time.Now()
        result.ExpiresAt = result.CachedAt.Add(c.config.CacheTTL)
        c.cache.Set(cacheKey, result)

        return hasAccess, nil
}

// requestPermissionWithFallback handles the API request with circuit breaker and fallback logic.
func (c *Client) requestPermissionWithFallback(ctx context.Context, userID, repoOwner, repoName string, operation Operation, cachedResult *PermissionResult) (bool, error) {
        // If circuit breaker is disabled, use regular request logic
        if c.circuitBreaker == nil {
                hasAccess, err := c.requestPermission(ctx, userID, repoOwner, repoName, operation)
                if err != nil {
                        // Fall back to cached result if available
                        if cachedResult != nil {
                                return cachedResult.HasAccess, nil
                        }
                        // Apply fallback policy
                        fallbackAccess := c.config.FallbackPolicy.ShouldAllowAccess(operation, false)
                        return fallbackAccess, &ServiceUnavailableError{
                                Operation:      operation,
                                UserID:         userID,
                                RepoOwner:      repoOwner,
                                RepoName:       repoName,
                                FallbackUsed:   true,
                                FallbackAccess: fallbackAccess,
                                Err:            err,
                        }
                }
                return hasAccess, nil
        }

        // Use circuit breaker for API requests
        var requestErr error
        var hasAccess bool

        err := c.circuitBreaker.Execute(ctx, func(ctx context.Context) error {
                var err error
                hasAccess, err = c.requestPermission(ctx, userID, repoOwner, repoName, operation)
                requestErr = err
                return err
        })

        // Handle circuit breaker states
        if err != nil {
                var cbErr *CircuitBreakerError
                if errors.As(err, &cbErr) {
                        // Circuit breaker is open - apply fallback policy
                        hasCachedResult := cachedResult != nil
                        if hasCachedResult {
                                return cachedResult.HasAccess, nil
                        }

                        fallbackAccess := c.config.FallbackPolicy.ShouldAllowAccess(operation, hasCachedResult)
                        return fallbackAccess, &ServiceUnavailableError{
                                Operation:      operation,
                                UserID:         userID,
                                RepoOwner:      repoOwner,
                                RepoName:       repoName,
                                CircuitState:   cbErr.State,
                                FallbackUsed:   true,
                                FallbackAccess: fallbackAccess,
                                Err:            cbErr,
                        }
                }

                // Other errors from the request - apply fallback logic
                if cachedResult != nil {
                        return cachedResult.HasAccess, nil
                }

                fallbackAccess := c.config.FallbackPolicy.ShouldAllowAccess(operation, false)
                return fallbackAccess, &ServiceUnavailableError{
                        Operation:      operation,
                        UserID:         userID,
                        RepoOwner:      repoOwner,
                        RepoName:       repoName,
                        FallbackUsed:   true,
                        FallbackAccess: fallbackAccess,
                        Err:            requestErr,
                }
        }

        return hasAccess, nil
}

// requestPermission makes an HTTP request to the API-Service to check permissions.
func (c *Client) requestPermission(ctx context.Context, userID, repoOwner, repoName string, operation Operation) (bool, error) {
        // Create backoff strategy
        backoffStrategy := backoff.NewExponentialBackOff()
        backoffStrategy.InitialInterval = c.config.InitialRetryDelay
        backoffStrategy.MaxInterval = c.config.MaxRetryDelay
        backoffStrategy.MaxElapsedTime = 0 // No overall timeout; rely on context

        // Limit the number of retries
        retryBackoff := backoff.WithMaxRetries(backoffStrategy, uint64(c.config.MaxRetries))

        // Add context cancellation to backoff
        retryBackoff = backoff.WithContext(retryBackoff, ctx)

        var lastErr error

        // Retry operation
        retryOperation := func() error {
                hasAccess, err := c.makeHTTPRequest(ctx, userID, repoOwner, repoName, operation)
                if err != nil {
                        lastErr = err

                        // Check if error is retryable
                        var permErr *PermissionError
                        if errors.As(err, &permErr) && !permErr.IsRetryable() {
                                // Non-retryable error, stop retrying
                                return backoff.Permanent(err)
                        }

                        // Retryable error, continue retrying
                        return err
                }

                // Success - store result and return
                lastErr = nil

                // Cache the successful result
                result := &PermissionResult{
                        UserID:    userID,
                        RepoOwner: repoOwner,
                        RepoName:  repoName,
                        Operation: operation,
                        HasAccess: hasAccess,
                        CachedAt:  time.Now(),
                        ExpiresAt: time.Now().Add(c.config.CacheTTL),
                }
                c.cache.Set(result.CacheKey(), result)

                return nil
        }

        err := backoff.Retry(retryOperation, retryBackoff)
        if err != nil {
                if lastErr != nil {
                        return false, lastErr
                }
                return false, err
        }

        // If we reach here, the operation succeeded
        // Get the cached result (which was set in the retry operation)
        result := c.cache.Get((&PermissionResult{
                UserID:    userID,
                RepoOwner: repoOwner,
                RepoName:  repoName,
                Operation: operation,
        }).CacheKey())

        if result != nil {
                return result.HasAccess, nil
        }

        // This should not happen, but fallback to false
        return false, errors.New("permission check completed but result not found in cache")
}

// makeHTTPRequest performs a single HTTP request to the API-Service.
func (c *Client) makeHTTPRequest(ctx context.Context, userID, repoOwner, repoName string, operation Operation) (bool, error) {
        // Build URL: GET http://api-service.local/permissions/:user_id/:repo_owner/:repo_name?op=read|write
        apiURL, err := c.buildAPIURL(userID, repoOwner, repoName, operation)
        if err != nil {
                return false, NewPermissionError(operation, userID, repoOwner, repoName,
                        fmt.Errorf("failed to build API URL: %w", err))
        }

        // Create HTTP request
        req, err := http.NewRequestWithContext(ctx, http.MethodGet, apiURL, nil)
        if err != nil {
                return false, NewPermissionError(operation, userID, repoOwner, repoName,
                        fmt.Errorf("failed to create HTTP request: %w", err))
        }

        // Set headers
        req.Header.Set("Content-Type", "application/json")
        req.Header.Set("User-Agent", "CollabOps-Git-Executor/1.0")

        // Make the request
        resp, err := c.httpClient.Do(req)
        if err != nil {
                // Network/connection errors are retryable
                return false, NewPermissionError(operation, userID, repoOwner, repoName,
                        fmt.Errorf("HTTP request failed: %w", err))
        }
        defer func() { _ = resp.Body.Close() }()

        // Read response body
        body, err := io.ReadAll(resp.Body)
        if err != nil {
                return false, NewPermissionErrorWithStatus(operation, userID, repoOwner, repoName, resp.StatusCode,
                        fmt.Errorf("failed to read response body: %w", err))
        }

        // Handle HTTP status codes
        if resp.StatusCode != http.StatusOK {
                err := fmt.Errorf("API returned status %d: %s", resp.StatusCode, string(body))
                return false, NewPermissionErrorWithStatus(operation, userID, repoOwner, repoName, resp.StatusCode, err)
        }

        // Parse JSON response
        var apiResponse APIResponse
        if err := json.Unmarshal(body, &apiResponse); err != nil {
                return false, NewPermissionErrorWithStatus(operation, userID, repoOwner, repoName, resp.StatusCode,
                        fmt.Errorf("failed to parse JSON response: %w", err))
        }

        return apiResponse.Allowed, nil
}

// buildAPIURL constructs the API URL for permission checking.
func (c *Client) buildAPIURL(userID, repoOwner, repoName string, operation Operation) (string, error) {
        // Parse base URL
        baseURL, err := url.Parse(c.config.APIServiceURL)
        if err != nil {
                return "", fmt.Errorf("invalid API service URL: %w", err)
        }

        // URL escape the path components
        escapedUserID := url.PathEscape(userID)
        escapedRepoOwner := url.PathEscape(repoOwner)
        escapedRepoName := url.PathEscape(repoName)

        // Build the path
        path := fmt.Sprintf("/permissions/%s/%s/%s", escapedUserID, escapedRepoOwner, escapedRepoName)

        // Build full URL with proper relative resolution
        fullURL := baseURL.ResolveReference(&url.URL{
                Path:     path,
                RawQuery: "op=" + url.QueryEscape(string(operation)),
        })

        return fullURL.String(), nil
}

// InvalidateCache implements PermissionClient.InvalidateCache.
func (c *Client) InvalidateCache(userID, repoOwner, repoName string) {
        readKey := userID + ":" + repoOwner + ":" + repoName + ":" + string(OperationRead)
        writeKey := userID + ":" + repoOwner + ":" + repoName + ":" + string(OperationWrite)
        c.cache.Delete(readKey)
        c.cache.Delete(writeKey)
}

// ClearCache implements PermissionClient.ClearCache.
func (c *Client) ClearCache() {
        c.cache.Clear()
}

func (c *Client) GetCircuitBreakerMetrics() *CircuitBreakerMetrics {
        if c.circuitBreaker == nil {
                return nil
        }
        metrics := c.circuitBreaker.GetMetrics()
        return &metrics
}

        c.cache.Close()
        return nil
}

// isRetryableStatusCode determines if an HTTP status code indicates a retryable error.
func isRetryableStatusCode(statusCode int) bool {
        switch statusCode {
        case http.StatusTooManyRequests, // 429
                http.StatusInternalServerError, // 500
                http.StatusBadGateway,          // 502
                http.StatusServiceUnavailable,  // 503
                http.StatusGatewayTimeout:      // 504
                return true
        case http.StatusUnauthorized, // 401
                http.StatusForbidden,  // 403
                http.StatusNotFound,   // 404
                http.StatusBadRequest: // 400
                return false
        default:
                // For unknown status codes >= 500, assume retryable
                return statusCode >= 500
        }
}

// WriteRefsWithCapabilities writes references with capability advertisement
func (cn *CapabilityNegotiator) WriteRefsWithCapabilities(w io.Writer, refs []Reference) error {
        pktWriter := newPktLineWriter(w)

        if len(refs) == 0 {
                // No refs available - write capabilities on null ref
                capLine := fmt.Sprintf("0000000000000000000000000000000000000000 capabilities^{}\x00%s\n", cn.serverCaps.String())
                if err := pktWriter.writeLine(capLine); err != nil {
                        return fmt.Errorf("failed to write capabilities: %w", err)
                }
        } else {
                // Write first ref with capabilities
                firstRef := refs[0]
                capLine := fmt.Sprintf("%s %s\x00%s\n", firstRef.Hash, firstRef.Name, cn.serverCaps.String())
                if err := pktWriter.writeLine(capLine); err != nil {
                        return fmt.Errorf("failed to write first ref with capabilities: %w", err)
                }

                // Write remaining refs without capabilities
                for _, ref := range refs[1:] {
                        refLine := fmt.Sprintf("%s %s\n", ref.Hash, ref.Name)
                        if err := pktWriter.writeLine(refLine); err != nil {
                                return fmt.Errorf("failed to write ref %s: %w", ref.Name, err)
                        }
                }
        }

        // End with flush packet
        if err := pktWriter.flush(); err != nil {
                return fmt.Errorf("failed to write flush packet: %w", err)
        }

        // Log capability advertisement
        if cn.logger != nil {
                cn.logger.LogProtocol("capability_advertisement", map[string]interface{}{
                        "capabilities": cn.serverCaps.String(),
                        "refs_count":   len(refs),
                        "symref":       cn.serverCaps.SymRef,
                })
        }

        return nil
}

// ParseClientCapabilities parses client capabilities from an upload-pack request
func (cn *CapabilityNegotiator) ParseClientCapabilities(body io.Reader) (*ClientCapabilities, error) {
        clientCaps := NewClientCapabilities()

        // Read and parse pkt-line format
        for {
                // Read packet length
                lengthBuf := make([]byte, 4)
                n, err := io.ReadFull(body, lengthBuf)
                if err != nil {
                        if err == io.EOF {
                                break // End of input
                        }
                        return nil, fmt.Errorf("failed to read packet length: %w", err)
                }
                if n != 4 {
                        break
                }

                // Parse length
                lengthStr := string(lengthBuf)
                if lengthStr == "0000" {
                        // Flush packet - end of section
                        break
                }

                // Convert hex length to integer
                var length int
                if _, err := fmt.Sscanf(lengthStr, "%04x", &length); err != nil {
                        return nil, fmt.Errorf("invalid packet length: %s", lengthStr)
                }

                if length < 4 {
                        return nil, fmt.Errorf("packet too short: %d", length)
                }

                // Read packet data
                dataLen := length - 4
                dataBuf := make([]byte, dataLen)
                n, err = io.ReadFull(body, dataBuf)
                if err != nil {
                        return nil, fmt.Errorf("failed to read packet data: %w", err)
                }
                if n != dataLen {
                        return nil, fmt.Errorf("incomplete packet data")
                }

                line := string(dataBuf)
                if err := cn.parsePacketLine(line, clientCaps); err != nil {
                        return nil, fmt.Errorf("failed to parse packet line: %w", err)
                }
        }

        // Log parsed capabilities
        if cn.logger != nil {
                cn.logger.LogProtocol("client_capabilities_parsed", map[string]interface{}{
                        "capabilities": clientCaps.Capabilities,
                        "wanted_refs":  len(clientCaps.WantedRefs),
                        "has_refs":     len(clientCaps.HasRefs),
                        "done":         clientCaps.Done,
                })
        }

        return clientCaps, nil
}

// parsePacketLine parses a single packet line and updates client capabilities
func (cn *CapabilityNegotiator) parsePacketLine(line string, clientCaps *ClientCapabilities) error {
        line = strings.TrimSuffix(line, "\n")

        if strings.HasPrefix(line, "want ") {
                parts := strings.Fields(line)
                if len(parts) < 2 {
                        return fmt.Errorf("invalid want line: %s", line)
                }

                hash := parts[1]
                clientCaps.WantedRefs = append(clientCaps.WantedRefs, hash)

                // Check for capabilities (after null byte)
                if strings.Contains(line, "\x00") {
                        capParts := strings.Split(line, "\x00")
                        if len(capParts) > 1 {
                                capStr := capParts[1]
                                if err := cn.parseCapabilityString(capStr, clientCaps); err != nil {
                                        return fmt.Errorf("failed to parse capabilities: %w", err)
                                }
                        }
                }
        } else if strings.HasPrefix(line, "have ") {
                parts := strings.Fields(line)
                if len(parts) < 2 {
                        return fmt.Errorf("invalid have line: %s", line)
                }

                hash := parts[1]
                clientCaps.HasRefs = append(clientCaps.HasRefs, hash)
        } else if line == "done" {
                clientCaps.Done = true
        }

        return nil
}

// parseCapabilityString parses a capability string like "multi_ack thin-pack agent=git/2.39.0"
func (cn *CapabilityNegotiator) parseCapabilityString(capStr string, clientCaps *ClientCapabilities) error {
        caps := strings.Fields(capStr)

        for _, cap := range caps {
                if strings.Contains(cap, "=") {
                        // Capability with value
                        parts := strings.SplitN(cap, "=", 2)
                        clientCaps.Capabilities[parts[0]] = parts[1]
                } else {
                        // Boolean capability
                        clientCaps.Capabilities[cap] = ""
                }
        }

        return nil
}

func (cn *CapabilityNegotiator) ValidateClientCapabilities(clientCaps *ClientCapabilities) error {
        // Check for required capabilities or incompatible combinations

        // If client wants thin-pack, server must support it
        if clientCaps.HasCapability("thin-pack") && !cn.serverCaps.ThinPack {
                return NewGitProtocolError("unsupported_capability",
                        "Client requested thin-pack but server does not support it",

                        "Client requested side-band-64k but server does not support it",

                cn.logger.LogProtocol("capability_validation", map[string]interface{}{
                        "client_capabilities": clientCaps.Capabilities,
                        "validation_result":   "passed",
                })
        }

        return nil
}

// HandleError handles an error according to Git Smart HTTP protocol requirements
func (eh *ErrorHandler) HandleError(w http.ResponseWriter, r *http.Request, err error, operation string) {
        var gitErr *GitProtocolError

        // Try to cast to GitProtocolError
        if e, ok := err.(*GitProtocolError); ok {
                gitErr = e
        } else {
                // Wrap unknown errors as internal server errors
                gitErr = &GitProtocolError{
                        Type:        "unknown_error",
                        Message:     err.Error(),
                        HTTPStatus:  http.StatusInternalServerError,
                        ProtocolErr: false,
                }
        }

        // Log the error with context
        if eh.logger != nil {
                details := map[string]interface{}{
                        "error_type":     gitErr.Type,
                        "http_status":    gitErr.HTTPStatus,
                        "protocol_error": gitErr.ProtocolErr,
                        "path":           r.URL.Path,
                        "method":         r.Method,
                        "user_id":        r.Header.Get("X-USER-ID"),
                        "request_id":     r.Header.Get("X-REQUEST-ID"),
                }
                eh.logger.LogError(operation, gitErr.Type, gitErr.Message, details)
        }

        // Send appropriate response
        if gitErr.ProtocolErr {
                eh.writeProtocolError(w, gitErr)
        } else {
                eh.writeHTTPError(w, gitErr)
        }
}

// writeProtocolError writes a Git protocol-compliant error response
func (eh *ErrorHandler) writeProtocolError(w http.ResponseWriter, err *GitProtocolError) {
        // Set appropriate content type for Git protocol
        w.Header().Set("Content-Type", "application/x-git-upload-pack-result")
        w.Header().Set("Cache-Control", "no-cache")
        w.WriteHeader(err.HTTPStatus)

        // Write error in pkt-line format
        pktWriter := newPktLineWriter(w)
        errorLine := fmt.Sprintf("ERR %s\n", err.Message)
        _ = pktWriter.writeLine(errorLine)
        _ = pktWriter.flush() // Send flush packet to end
}

// ValidateContentType validates that the request has an appropriate content type for Git Smart HTTP
func ValidateContentType(r *http.Request, expectedTypes []string) error {
        if r.Method != http.MethodPost {
                return nil // GET requests don't require specific content types
        }

        contentType := r.Header.Get("Content-Type")
        if contentType == "" {
                return NewGitProtocolError("missing_content_type", "Missing Content-Type header", http.StatusBadRequest, false)
        }

        // Remove charset and other parameters
        contentType = strings.Split(contentType, ";")[0]
        contentType = strings.TrimSpace(contentType)

        for _, expected := range expectedTypes {
                if contentType == expected {
                        return nil
                }
        }

        return NewGitProtocolError("invalid_content_type",
                fmt.Sprintf("Invalid content type: %s, expected one of: %s", contentType, strings.Join(expectedTypes, ", ")),
                http.StatusBadRequest, false)
}

// WrapRepositoryError wraps repository-related errors into appropriate Git protocol errors
func WrapRepositoryError(err error, repoPath string) error {
        if err == nil {
                return nil
        }

        errMsg := err.Error()

        // Check for common repository error patterns
        if strings.Contains(errMsg, "repository does not exist") ||
                strings.Contains(errMsg, "no such file or directory") ||
                strings.Contains(errMsg, "not found") {
                return NewGitProtocolError("repository_not_found",
                        fmt.Sprintf("Repository not found: %s", repoPath),
                        http.StatusNotFound, false)
        }

        if strings.Contains(errMsg, "permission denied") ||
                strings.Contains(errMsg, "access denied") {
                return NewGitProtocolError("repository_access_denied",
                        fmt.Sprintf("Access denied to repository: %s", repoPath),
                        http.StatusForbidden, false)
        }

        if strings.Contains(errMsg, "invalid repository") ||
                strings.Contains(errMsg, "not a git repository") {
                return NewGitProtocolError("invalid_repository",
                        fmt.Sprintf("Invalid repository: %s", repoPath),
                        http.StatusBadRequest, false)
        }

        // Default to internal server error for unknown repository errors
        return NewGitProtocolError("repository_error",
                fmt.Sprintf("Repository error: %s", errMsg),
                http.StatusInternalServerError, false)
}

// WrapPermissionError wraps permission-related errors into appropriate Git protocol errors
func WrapPermissionError(err error, operation string) error {

        if strings.Contains(errMsg, "permission denied") ||

                return ErrPermissionDenied
        }

        }

        // Default to internal server error for unknown permission errors
        return NewGitProtocolError("permission_check_failed",
                fmt.Sprintf("Permission check failed: %s", errMsg),
                http.StatusInternalServerError, false)
}

// HandleInfoRefs handles GET /repos/:owner/:repo/info/refs?service=git-upload-pack
// This endpoint advertises references and capabilities to the gateway (which forwards to client)
func (h *UploadPackHandler) HandleInfoRefs(w http.ResponseWriter, r *http.Request) {
        ctx := r.Context()

        // Extract path parameters from gateway-forwarded request
        owner := chi.URLParam(r, "owner")
        repo := chi.URLParam(r, "repo")
        service := r.URL.Query().Get("service")

        // Validate service parameter
        if service != "git-upload-pack" {
                http.Error(w, "Invalid service parameter", http.StatusBadRequest)
                return
        }

        // Get user ID from gateway context (forwarded header)
        userID := r.Header.Get("X-USER-ID")
        if userID == "" {
                http.Error(w, "Missing user authentication context", http.StatusUnauthorized)
                return
        }

        // Check user permissions for read access
        hasAccess, err := h.permissionClient.CheckReadAccess(ctx, userID, owner, repo)
        if err != nil {
                http.Error(w, fmt.Sprintf("Permission check failed: %v", err), http.StatusInternalServerError)
                return
        }
        if !hasAccess {
                http.Error(w, "Access denied", http.StatusForbidden)
                return
        }

        // Get repository path and validate
        repoPath, err := repository.GetAndValidateRepoPath(h.pathConfig, owner, repo)
        if err != nil {
                http.Error(w, fmt.Sprintf("Invalid repository path: %v", err), http.StatusBadRequest)
                return
        }

        // Ensure repository exists
        if err := repository.EnsureRepoExists(h.pathConfig, repoPath); err != nil {
                http.Error(w, "Repository not found", http.StatusNotFound)
                return
        }

        // Open repository using go-git with caching
        gitRepo, err := git.OpenRepositoryWithCache(repoPath)
        if err != nil {
                http.Error(w, fmt.Sprintf("Failed to open repository: %v", err), http.StatusInternalServerError)
                return
        }
        defer func() { _ = gitRepo.Close() }()

        // Set response headers for Git Smart HTTP protocol
        w.Header().Set("Content-Type", "application/x-git-upload-pack-advertisement")
        w.Header().Set("Cache-Control", "no-cache")

        // Write service advertisement
        _, _ = fmt.Fprintf(w, "001e# service=git-upload-pack\n")
        _, _ = fmt.Fprintf(w, "0000") // flush packet

        // Get repository references using our git package
        // For now, we'll implement a basic version that gets HEAD and branches
        err = h.writeReferences(ctx, w, gitRepo)
        if err != nil {
                // If we've already started writing, we can't send an HTTP error
                // Log the error and write an error packet instead
                _, _ = fmt.Fprintf(w, "0000") // end packet
                return
        }

        // End with null packet
        _, _ = fmt.Fprintf(w, "0000")
}

        w.Header().Set("Content-Type", "application/x-git-upload-pack-result")

// parseUploadPackRequest parses the upload-pack request body in pkt-line format
func (h *UploadPackHandler) parseUploadPackRequest(body io.Reader) (*UploadPackRequest, error) {
        req := &UploadPackRequest{
                Wants: make([]string, 0),
                Haves: make([]string, 0),
        }

        buf := make([]byte, 4) // Buffer for reading length header
        firstWant := true

        for {
                // Read 4-byte length header
                n, err := io.ReadFull(body, buf)
                if err != nil {
                        if err == io.EOF {
                                break // End of input
                        }
                        return req, err
                }
                if n != 4 {
                        break
                }

                // Parse length
                lengthHex := string(buf)
                length, err := strconv.ParseInt(lengthHex, 16, 32)
                if err != nil {
                        continue // Skip invalid packet
                }

                if length == 0 {
                        // Flush packet (0000)
                        continue
                }

                if length < 4 {
                        // Invalid packet length
                        continue
                }

                // Read the data portion (length - 4 for the header we already read)
                dataLen := int(length) - 4
                if dataLen <= 0 {
                        continue
                }

                dataBuf := make([]byte, dataLen)
                n, err = io.ReadFull(body, dataBuf)
                if err != nil {
                        if err == io.EOF {
                                break
                        }
                        return req, err
                }
                if n != dataLen {
                        continue
                }

                // Convert to string and trim newline if present
                data := strings.TrimSuffix(string(dataBuf), "\n")

                if strings.HasPrefix(data, "want ") {
                        parts := strings.Fields(data)
                        if len(parts) >= 2 {
                                req.Wants = append(req.Wants, parts[1])

                                // Capabilities are sent with the first want line
                                if firstWant && len(parts) > 2 {
                                        req.Capabilities = parts[2:]
                                        firstWant = false
                                }
                        }
                } else if strings.HasPrefix(data, "have ") {
                        parts := strings.Fields(data)
                        if len(parts) >= 2 {
                                req.Haves = append(req.Haves, parts[1])
                        }
                } else if strings.TrimSpace(data) == "done" {
                        req.Done = true
                        break
                }
        }

        return req, nil
}

// processWantHaveNegotiation handles the want/have negotiation and packfile generation
func (h *UploadPackHandler) processWantHaveNegotiation(ctx context.Context, repo *gogit.Repository, req *UploadPackRequest, pktWriter *pktLineWriter) error {
        // For now, implement a simple version that doesn't do complex negotiation
        // A full implementation would check for common commits and optimize the packfile

        if len(req.Wants) == 0 {
                return fmt.Errorf("no wants specified")
        }

        // Check if all wanted objects exist
        for _, want := range req.Wants {
                hash := plumbing.NewHash(want)
                _, err := repo.Object(plumbing.AnyObject, hash)
                if err != nil {
                        return fmt.Errorf("object %s not found: %w", want, err)
                }
        }

        // Send NAK (no acknowledgment) - in a full implementation, we'd check for common commits
        if err := pktWriter.writeLine("NAK"); err != nil {
                return fmt.Errorf("failed to write NAK: %w", err)
        }

        // Generate and stream a simplified packfile
        // Note: This is a basic implementation. A production version would generate
        // proper packfiles with object deltas and compression
        err := h.generateAndStreamPackfile(ctx, repo, req, pktWriter)
        if err != nil {
                return fmt.Errorf("failed to generate packfile: %w", err)
        }

        return nil
}

// generateAndStreamPackfile generates and streams a packfile for the requested objects
func (h *UploadPackHandler) generateAndStreamPackfile(ctx context.Context, repo *gogit.Repository, req *UploadPackRequest, pktWriter *pktLineWriter) error {
        // Send sideband packet indicating packfile data follows
        if err := pktWriter.writeLine("0008NAK\n"); err != nil {
                return err
        }

        // Flush to end the negotiation phase
        if err := pktWriter.flush(); err != nil {
                return err
        }

        // For this basic implementation, we'll collect the objects and send them
        // In a production system, this would generate a proper Git packfile format
        objectsToSend := make(map[plumbing.Hash]*object.Commit)

        // Collect all commits reachable from wants
        for _, want := range req.Wants {
                hash := plumbing.NewHash(want)
                commit, err := repo.CommitObject(hash)
                if err != nil {
                        continue // Skip if not a commit object
                }

                // Walk the commit history to collect all reachable commits
                err = h.walkCommitHistory(repo, commit, objectsToSend, req.Haves)
                if err != nil {
                        return fmt.Errorf("failed to walk commit history: %w", err)
                }
        }

        // Send a minimal "packfile" (this is simplified - real packfiles are binary format)
        packfileHeader := fmt.Sprintf("# pack with %d objects\n", len(objectsToSend))
        if err := pktWriter.writeLine(packfileHeader); err != nil {
                return err
        }

        // Send object information (simplified)
        for hash, commit := range objectsToSend {
                objectInfo := fmt.Sprintf("commit %s %s\n", hash.String(), commit.Message)
                if err := pktWriter.writeLine(objectInfo); err != nil {
                        return err
                }
        }

        // End with flush packet
        return pktWriter.flush()
}

// walkCommitHistory walks the commit history and collects commits
func (h *UploadPackHandler) walkCommitHistory(repo *gogit.Repository, commit *object.Commit, collected map[plumbing.Hash]*object.Commit, haves []string) error {
        hash := commit.Hash

        // Check if we've already processed this commit
        if _, exists := collected[hash]; exists {
                return nil
        }

        // Check if this commit is in the "have" list (client already has it)
        for _, have := range haves {
                if hash.String() == have {
                        return nil // Don't include objects the client already has
                }
        }

        // Add this commit to our collection
        collected[hash] = commit

        // Recursively walk parent commits (limit depth to prevent infinite loops)
        if len(collected) < 1000 { // Safety limit
                parents := commit.Parents()
                for {
                        parent, err := parents.Next()
                        if err != nil {
                                break // No more parents
                        }

                        err = h.walkCommitHistory(repo, parent, collected, haves)
                        if err != nil {
                                return err
                        }
                }
        }

        return nil
}

// writeLine writes a packet line with the data
func (p *pktLineWriter) writeLine(data string) error {
        // Handle empty data as flush packet (Git protocol compliance)
        if data == "" {
                return p.flush()
        }

        // Add newline if not present (for normal Git protocol lines)
        if !strings.HasSuffix(data, "\n") {
                data += "\n"
        }

        // Calculate packet length (4 bytes for length + data)
        length := len(data) + 4

        // Check maximum packet line length (Git protocol limit)
        if length > 65520 {
                return fmt.Errorf("packet line too long: %d bytes (max 65520)", length)
        }

        line := fmt.Sprintf("%04x%s", length, data)

        _, err := p.w.Write([]byte(line))
        return err
}

// flush writes a flush packet (0000)
func (p *pktLineWriter) flush() error {
        _, err := p.w.Write([]byte("0000"))
        return err
}

        repoPath, err := h.getValidatedRepoPath(owner, repo)
        if err != nil {
                http.Error(w, fmt.Sprintf("Invalid repository path: %v", err), http.StatusBadRequest)
                return
        }

        // Open repository using go-git
        gitRepo, err := gogit.PlainOpen(repoPath)
        if err != nil {
                http.Error(w, "Repository not found", http.StatusNotFound)
                return
        }

        // Parse the upload-pack request
        uploadReq, err := h.parseUploadPackRequest(r.Body)
        if err != nil {
                http.Error(w, fmt.Sprintf("Invalid request format: %v", err), http.StatusBadRequest)
                return
        }

        // Set response headers for Git Smart HTTP protocol
        w.Header().Set("Content-Type", "application/x-git-upload-pack-result")
        w.Header().Set("Cache-Control", "no-cache")

        // Create streaming request
        streamReq := &streaming.PackfileStreamRequest{
                Wants:        uploadReq.Wants,
                Haves:        uploadReq.Haves,
                Capabilities: uploadReq.Capabilities,
        }

        // Create Git streamer and stream the packfile
        streamer := streaming.NewGitStreamer(gitRepo)
        err = streamer.StreamPackfile(ctx, streamReq, w)
        if err != nil {
                // If we've already started writing, we can't send HTTP error anymore
                // The streaming mechanism will handle internal errors gracefully
                fmt.Printf("Streaming error: %v\n", err)
                return
        }
}

// HandleInfoRefsStreaming provides streaming-optimized info/refs functionality
func (h *StreamingUploadPackHandler) HandleInfoRefsStreaming(w http.ResponseWriter, r *http.Request) {
        ctx := r.Context()

        // Extract path parameters from gateway-forwarded request
        owner := chi.URLParam(r, "owner")
        repo := chi.URLParam(r, "repo")
        service := r.URL.Query().Get("service")

        // Validate service parameter
        if service != "git-upload-pack" {
                http.Error(w, "Invalid service parameter", http.StatusBadRequest)
                return
        }

        // Get user ID from gateway context (forwarded header)
        userID := r.Header.Get("X-USER-ID")
        if userID == "" {
                http.Error(w, "Missing user authentication context", http.StatusUnauthorized)
                return
        }

        // Check user permissions for read access
        hasAccess, err := h.permissionClient.CheckReadAccess(ctx, userID, owner, repo)
        if err != nil {
                http.Error(w, fmt.Sprintf("Permission check failed: %v", err), http.StatusInternalServerError)
                return
        }
        if !hasAccess {
                http.Error(w, "Access denied", http.StatusForbidden)
                return
        }

        // Get repository path and validate
        repoPath, err := h.getValidatedRepoPath(owner, repo)
        if err != nil {
                http.Error(w, fmt.Sprintf("Invalid repository path: %v", err), http.StatusBadRequest)
                return
        }

        // Open repository using go-git
        gitRepo, err := gogit.PlainOpen(repoPath)
        if err != nil {
                http.Error(w, "Repository not found", http.StatusNotFound)
                return
        }

        // Set response headers for Git Smart HTTP protocol
        w.Header().Set("Content-Type", "application/x-git-upload-pack-advertisement")
        w.Header().Set("Cache-Control", "no-cache")

        // Write service advertisement
        _, _ = fmt.Fprintf(w, "001e# service=git-upload-pack\n")
        _, _ = fmt.Fprintf(w, "0000") // flush packet

        // Create Git streamer and stream references
        streamer := streaming.NewGitStreamer(gitRepo)
        capabilities := []string{"multi_ack", "thin-pack", "side-band", "side-band-64k", "ofs-delta", "shallow", "no-progress", "include-tag", "multi_ack_detailed", "no-done", "symref=HEAD:refs/heads/main", "agent=git/collabops-git-executor"}

        err = streamer.StreamRefs(ctx, w, capabilities)
        if err != nil {
                // If we've already started writing, we can't send HTTP error anymore
                fmt.Printf("Reference streaming error: %v\n", err)
                return
        }

        // End with null packet
        _, _ = fmt.Fprintf(w, "0000")
}

        // Use the existing path validation from the base handler

}

// Global configuration management with thread safety

var (
        // globalConfig holds the global default configuration
        globalConfig     *PathConfig
        globalConfigOnce sync.Once
        globalConfigMux  sync.RWMutex
)

// SetGlobalConfig sets the global default configuration for the package.
// This function is thread-safe and can be called from multiple goroutines.
// The configuration will be used by all *WithDefault functions.
func SetGlobalConfig(config *PathConfig) error {
        if config == nil {
                return &ValidationError{
                        Component: "global_config",
                        Reason:    "configuration cannot be nil",
                }
        }

        globalConfigMux.Lock()
        defer globalConfigMux.Unlock()
        globalConfig = config
        return nil
}

// GetGlobalConfig returns the current global configuration.
// This function is thread-safe and returns a copy to prevent modification.
// If no global configuration has been set, it returns the default configuration.
func GetGlobalConfig() (*PathConfig, error) {
        globalConfigMux.RLock()
        defer globalConfigMux.RUnlock()

        if globalConfig != nil {
                // Return a copy to prevent external modification
                return &PathConfig{RepoRoot: globalConfig.RepoRoot}, nil
        }

        // Initialize default configuration lazily
        var err error
        globalConfigOnce.Do(func() {
                defaultConfig, configErr := NewPathConfig(DefaultRepoRoot)
                if configErr != nil {
                        err = &PathError{
                                Op:  "initialize_global_config",
                                Err: configErr,
                        }
                        return
                }
                globalConfig = defaultConfig
        })

        if err != nil {
                return nil, err
        }

        // Return a copy of the default configuration
        return &PathConfig{RepoRoot: globalConfig.RepoRoot}, nil
}

// ResetGlobalConfig resets the global configuration to nil.
// This is primarily useful for testing scenarios.
// This function is thread-safe.
func ResetGlobalConfig() {
        globalConfigMux.Lock()
        defer globalConfigMux.Unlock()
        globalConfig = nil
        globalConfigOnce = sync.Once{}
}

        RepoExtension = ".git"

        // PathSeparator is the standard path separator for repository paths
        PathSeparator = "/"

// PathConfig holds configuration for repository path management
type PathConfig struct {
        // RepoRoot is the base directory where all repositories are stored
        RepoRoot string

// NewPathConfig creates a new PathConfig with validation
func NewPathConfig(repoRoot string) (*PathConfig, error) {
        if repoRoot == "" {
                return nil, fmt.Errorf("repository root path cannot be empty")
        }

        // Clean the path to ensure consistent formatting
        cleanRoot := filepath.Clean(repoRoot)

        // Ensure the path is absolute
        if !filepath.IsAbs(cleanRoot) {
                return nil, fmt.Errorf("repository root path must be absolute: %s", cleanRoot)
        }

        // Resolve symlinks to ensure consistent path comparison (fixes macOS /var -> /private/var issue)
        realRoot, err := filepath.EvalSymlinks(cleanRoot)
        if err != nil {
                // If symlink resolution fails, use the original clean path
                realRoot = cleanRoot
        }

        return &PathConfig{
                RepoRoot: realRoot,
        }, nil
}

// GetRepoRoot returns the configured repository root path
func (pc *PathConfig) GetRepoRoot() string {
        return pc.RepoRoot
}

// FormatRepoPath generates the expected path pattern for a repository
// Format: {RepoRoot}/{owner}/{repo}.git
func (pc *PathConfig) FormatRepoPath(owner, repo string) string {
        return filepath.Join(pc.RepoRoot, owner, repo+RepoExtension)
}

// GetRepoPath returns the filesystem path for a given repository
// This function validates the owner and repo components and constructs the full path
func GetRepoPath(pathConfig *PathConfig, owner, repo string) (string, error) {
        // Validate path components for security
        if err := ValidatePathComponents(owner, repo); err != nil {
                return "", fmt.Errorf("invalid path components: %w", err)
        }

        // Generate the repository path
        repoPath := pathConfig.FormatRepoPath(owner, repo)

        return repoPath, nil
}

// when no PathConfig is provided
func GetRepoPathWithDefault(owner, repo string) (string, error) {

        if err != nil {
                return "", fmt.Errorf("failed to create default path config: %w", err)
        }

        return GetRepoPath(defaultConfig, owner, repo)
}

// ValidateRepoPath validates if a given path is within the allowed repository root
// and prevents directory traversal attacks
func ValidateRepoPath(pathConfig *PathConfig, repoPath string) error {
        if pathConfig == nil {
                return fmt.Errorf("path configuration cannot be nil")
        }

        if repoPath == "" {
                return fmt.Errorf("repository path cannot be empty")
        }

        // Clean the provided path to resolve any relative components
        cleanPath := filepath.Clean(repoPath)

        // Convert to absolute path to handle cases where relative paths are passed
        absPath, err := filepath.Abs(cleanPath)
        if err != nil {
                return fmt.Errorf("failed to resolve absolute path for %s: %w", repoPath, err)
        }

        // Resolve symlinks to ensure consistent path comparison (fixes macOS /var -> /private/var issue)
        realPath, err := filepath.EvalSymlinks(absPath)
        if err != nil {
                // If symlink resolution fails, use the original absolute path
                realPath = absPath
        }

        // The repository root should already be resolved to real path by NewPathConfig
        absRepoRoot := pathConfig.RepoRoot

        // Normalize paths for consistent comparison (handles case sensitivity on Windows)
        realPath = filepath.Clean(realPath)
        absRepoRoot = filepath.Clean(absRepoRoot)

        // Check if the path is within the repository root
        // Use filepath.Rel to check if the path is a subdirectory
        relPath, err := filepath.Rel(absRepoRoot, realPath)
        if err != nil {
                return fmt.Errorf("failed to determine relative path from %s to %s: %w", absRepoRoot, realPath, err)
        }

        // If the relative path starts with ".." or is ".", it's outside or equal to the root
        if strings.HasPrefix(relPath, "..") {
                return fmt.Errorf("repository path %s is outside the allowed repository root %s", repoPath, pathConfig.RepoRoot)
        }

        // Check for exact match to repository root (which we don't allow)
        if relPath == "." {
                return fmt.Errorf("repository path cannot be the repository root itself: %s", repoPath)
        }

        // Additional security check: ensure the path contains expected structure
        // A valid repository path should be at least 2 levels deep: root/owner/repo.git
        pathParts := strings.Split(relPath, string(filepath.Separator))
        if len(pathParts) < 2 {
                return fmt.Errorf("repository path %s does not follow expected structure (owner/repo.git)", repoPath)
        }

        // Validate that the path ends with .git extension
        if !strings.HasSuffix(realPath, RepoExtension) {
                return fmt.Errorf("repository path %s must end with %s extension", repoPath, RepoExtension)
        }

        return nil
}

// ValidateRepoPathWithDefault is a convenience function that uses the default configuration
// when no PathConfig is provided
func ValidateRepoPathWithDefault(repoPath string) error {
        defaultConfig, err := NewPathConfig(DefaultRepoRoot)
        if err != nil {
                return fmt.Errorf("failed to create default path config: %w", err)
        }

        return ValidateRepoPath(defaultConfig, repoPath)
}

// ValidatePathComponents validates owner and repository name components
func ValidatePathComponents(owner, repo string) error {
        if err := validateComponent(owner, "owner"); err != nil {
                return err
        }

        if err := validateComponent(repo, "repository"); err != nil {
                return err
        }

        return nil
}

// validateComponent validates a single path component (owner or repo name)
func validateComponent(component, componentType string) error {
        if component == "" {
                return fmt.Errorf("%s name cannot be empty", componentType)
        }

        // Check for path traversal attempts
        if strings.Contains(component, "..") {
                return fmt.Errorf("%s name cannot contain '..' (path traversal attempt): %s", componentType, component)
        }

        // Check for absolute path attempts
        if strings.HasPrefix(component, "/") {
                return fmt.Errorf("%s name cannot start with '/' (absolute path attempt): %s", componentType, component)
        }

        // Check for invalid characters (basic check for common problematic characters)
        invalidChars := []string{"\\", ":", "*", "?", "\"", "<", ">", "|"}
        for _, char := range invalidChars {
                if strings.Contains(component, char) {
                        return fmt.Errorf("%s name contains invalid character '%s': %s", componentType, char, component)
                }
        }

        // Check for reserved names on Windows (even though we're primarily Linux-focused)
        reservedNames := []string{"CON", "PRN", "AUX", "NUL", "COM1", "COM2", "COM3", "COM4", "COM5", "COM6", "COM7", "COM8", "COM9", "LPT1", "LPT2", "LPT3", "LPT4", "LPT5", "LPT6", "LPT7", "LPT8", "LPT9"}
        upperComponent := strings.ToUpper(component)
        for _, reserved := range reservedNames {
                if upperComponent == reserved {
                        return fmt.Errorf("%s name cannot be a reserved system name: %s", componentType, component)
                }
        }

        return nil
}

// EnsureRepoExists checks if a Git repository exists at the given path
// This function validates the path and checks for repository structure
func EnsureRepoExists(pathConfig *PathConfig, repoPath string) error {
        // First validate the path using our security validation
        if err := ValidateRepoPath(pathConfig, repoPath); err != nil {
                return fmt.Errorf("path validation failed: %w", err)
        }

        // Check if the path exists
        if err := checkPathExists(repoPath); err != nil {
                return err
        }

        // Check if it's a valid Git repository
        if err := validateGitRepository(repoPath); err != nil {
                return err
        }

        return nil
}

// EnsureRepoExistsWithDefault is a convenience function that uses the default configuration
func EnsureRepoExistsWithDefault(repoPath string) error {
        defaultConfig, err := NewPathConfig(DefaultRepoRoot)
        if err != nil {
                return fmt.Errorf("failed to create default path config: %w", err)
        }

        return EnsureRepoExists(defaultConfig, repoPath)
}

// checkPathExists verifies that the given path exists and is accessible
func checkPathExists(path string) error {
        if path == "" {
                return fmt.Errorf("repository path cannot be empty")
        }

        // Use os.Stat to check if path exists and get file info
        info, err := os.Stat(path)
        if err != nil {
                if os.IsNotExist(err) {
                        return fmt.Errorf("repository does not exist at path: %s", path)
                }
                if os.IsPermission(err) {
                        return fmt.Errorf("permission denied accessing repository at path: %s", path)
                }
                return fmt.Errorf("failed to access repository at path %s: %w", path, err)
        }

        // Ensure the path is a directory
        if !info.IsDir() {
                return fmt.Errorf("repository path is not a directory: %s", path)
        }

        return nil
}

// validateGitRepository checks if the path contains a valid Git repository structure
func validateGitRepository(repoPath string) error {
        // Check for bare repository (common for server-side repositories)
        if isBareRepository(repoPath) {
                return nil // Valid bare repository
        }

        // Check for non-bare repository (.git subdirectory)
        if isNonBareRepository(repoPath) {
                return nil // Valid non-bare repository
        }

        // Path exists but doesn't appear to be a Git repository
        return fmt.Errorf("path exists but does not contain a valid Git repository structure: %s", repoPath)
}

// isBareRepository checks if the given path is a bare Git repository
func isBareRepository(repoPath string) bool {
        // Bare repositories contain Git files directly in the root directory
        // Key indicators: HEAD file, refs/ directory, objects/ directory

        // Check for HEAD file
        headPath := filepath.Join(repoPath, "HEAD")
        if _, err := os.Stat(headPath); err != nil {
                return false
        }

        // Check for refs directory
        refsPath := filepath.Join(repoPath, "refs")
        if info, err := os.Stat(refsPath); err != nil || !info.IsDir() {
                return false
        }

        // Check for objects directory
        objectsPath := filepath.Join(repoPath, "objects")
        if info, err := os.Stat(objectsPath); err != nil || !info.IsDir() {
                return false
        }

        // Additional check: config file should exist
        configPath := filepath.Join(repoPath, "config")
        if _, err := os.Stat(configPath); err != nil {
                return false
        }

        return true
}

// isNonBareRepository checks if the given path contains a .git subdirectory
func isNonBareRepository(repoPath string) bool {
        // Non-bare repositories have a .git subdirectory containing Git metadata
        gitPath := filepath.Join(repoPath, ".git")

        if info, err := os.Stat(gitPath); err != nil {
                return false
        } else if info.IsDir() {
                // .git is a directory - check for key Git files inside
                return validateGitDirectory(gitPath)
        } else {
                // .git might be a file (worktree reference) - still valid but different structure
                return validateGitFile(gitPath)
        }
}

// validateGitDirectory checks if a .git directory contains required Git structure
func validateGitDirectory(gitPath string) bool {
        // Check for HEAD file
        headPath := filepath.Join(gitPath, "HEAD")
        if _, err := os.Stat(headPath); err != nil {
                return false
        }

        // Check for refs directory
        refsPath := filepath.Join(gitPath, "refs")
        if info, err := os.Stat(refsPath); err != nil || !info.IsDir() {
                return false
        }

        // Check for objects directory
        objectsPath := filepath.Join(gitPath, "objects")
        if info, err := os.Stat(objectsPath); err != nil || !info.IsDir() {
                return false
        }

        return true
}

// validateGitFile checks if a .git file contains a valid worktree reference
func validateGitFile(gitPath string) bool {
        // Read the .git file content
        content, err := os.ReadFile(gitPath)
        if err != nil {
                return false
        }

        // .git file should contain "gitdir: <path>" for worktrees
        contentStr := strings.TrimSpace(string(content))
        if !strings.HasPrefix(contentStr, "gitdir: ") {
                return false
        }

        // Extract the referenced git directory path
        gitDir := strings.TrimPrefix(contentStr, "gitdir: ")
        gitDir = strings.TrimSpace(gitDir)

        // If path is relative, make it relative to the .git file location
        if !filepath.IsAbs(gitDir) {
                gitDir = filepath.Join(filepath.Dir(gitPath), gitDir)
        }

        // Validate that the referenced directory exists and is a valid Git directory
        if info, err := os.Stat(gitDir); err != nil || !info.IsDir() {
                return false
        }

        return validateGitDirectory(gitDir)
}

// RepositoryInfo holds metadata about a repository
type RepositoryInfo struct {
        Owner        string // Repository owner
        Name         string // Repository name
        Path         string // Full filesystem path
        Type         string // Repository type ("bare", "non-bare", "worktree")
        Exists       bool   // Whether the repository exists
        IsValid      bool   // Whether the repository structure is valid
        IsAccessible bool   // Whether the repository is accessible (no permission issues)
}

// GetRepositoryInfo returns comprehensive information about a repository.
// This function performs all repository checks and returns detailed metadata.
func GetRepositoryInfo(pathConfig *PathConfig, owner, repo string) (*RepositoryInfo, error) {
        info := &RepositoryInfo{
                Owner: owner,
                Name:  repo,
        }

        // Generate repository path
        repoPath, err := GetRepoPath(pathConfig, owner, repo)
        if err != nil {
                return info, &PathError{
                        Op:  "get_repository_info",
                        Err: err,
                }
        }
        info.Path = repoPath

        // Validate path
        if err := ValidateRepoPath(pathConfig, repoPath); err != nil {
                return info, &PathError{
                        Op:   "get_repository_info",
                        Path: repoPath,
                        Err:  err,
                }
        }

        // Check repository existence and structure
        if err := EnsureRepoExists(pathConfig, repoPath); err != nil {
                // Check what kind of error this is by examining the error message or unwrapping
                errStr := err.Error()

                if strings.Contains(errStr, "repository does not exist at path") {
                        info.Exists = false
                        info.IsAccessible = false
                        info.IsValid = false
                } else if strings.Contains(errStr, "permission denied") {
                        info.Exists = true
                        info.IsAccessible = false
                        info.IsValid = false
                } else {
                        // Repository exists but structure is invalid
                        info.Exists = true
                        info.IsAccessible = true
                        info.IsValid = false
                }
                return info, nil
        }

        // Repository is fully valid and accessible
        info.Exists = true
        info.IsAccessible = true
        info.IsValid = true

        // Determine repository type
        if isBareRepository(repoPath) {
                info.Type = "bare"
        } else if isNonBareRepository(repoPath) {
                // Check if it's a worktree
                gitPath := filepath.Join(repoPath, ".git")
                if fileInfo, err := os.Stat(gitPath); err == nil && !fileInfo.IsDir() {
                        info.Type = "worktree"

        }

// Start initializes and starts the HTTP server
func (s *Server) Start() error {
        // Create router
        router := s.setupRouter()

        // Create HTTP server
        s.server = &http.Server{
                Addr:         fmt.Sprintf("%s:%d", s.config.Server.Host, s.config.Server.Port),
                Handler:      router,
                ReadTimeout:  s.config.Server.ReadTimeout,
                WriteTimeout: s.config.Server.WriteTimeout,
                IdleTimeout:  s.config.Server.IdleTimeout,
        }

        // Check if port is available
        listener, err := net.Listen("tcp", s.server.Addr)
        if err != nil {
                s.logger.Error("Failed to bind to address",
                        zap.String("address", s.server.Addr),
                        zap.Error(err))
                return fmt.Errorf("failed to bind to address %s: %w", s.server.Addr, err)
        }
        _ = listener.Close()

        // Start server in a goroutine
        go func() {
                s.logger.Info("Server starting",
                        zap.String("address", s.server.Addr),
                        zap.String("basePath", s.config.Server.BasePath))
                if err := s.server.ListenAndServe(); err != nil && err != http.ErrServerClosed {
                        s.logger.Error("Server failed to start", zap.Error(err))
                }
        }()

        // Wait for shutdown signal
        return s.waitForShutdown()
}

// waitForShutdown waits for shutdown signals and performs graceful shutdown
func (s *Server) waitForShutdown() error {
        // Create channel to listen for interrupt signals
        quit := make(chan os.Signal, 1)
        signal.Notify(quit, syscall.SIGINT, syscall.SIGTERM)

        // Block until a signal is received
        sig := <-quit
        s.logger.Info("Received shutdown signal", zap.String("signal", sig.String()))

        // Create shutdown context with timeout
        ctx, cancel := context.WithTimeout(context.Background(), s.config.Server.ShutdownTimeout)
        defer cancel()

        // Attempt graceful shutdown
        s.logger.Info("Shutting down server gracefully")
        if err := s.server.Shutdown(ctx); err != nil {
                s.logger.Error("Server forced to shutdown", zap.Error(err))
                return err
        }

        s.logger.Info("Server shutdown complete")
        return nil
}

// Shutdown gracefully shuts down the server
func (s *Server) Shutdown(ctx context.Context) error {
        if s.server == nil {
                return nil
        }
        return s.server.Shutdown(ctx)
}

// loggingMiddleware logs HTTP requests
func (s *Server) loggingMiddleware(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                start := time.Now()
                ww := middleware.NewWrapResponseWriter(w, r.ProtoMajor)

                next.ServeHTTP(ww, r)

                s.logger.Info("HTTP request",
                        zap.String("method", r.Method),
                        zap.String("path", r.URL.Path),
                        zap.Int("status", ww.Status()),
                        zap.Int("bytes", ww.BytesWritten()),
                        zap.Duration("duration", time.Since(start)),
                        zap.String("remote", r.RemoteAddr),
                )
        })
}

package streaming

import (

        "sync"

        gogit "github.com/go-git/go-git/v5"

// GitStreamer provides streaming functionality for Git operations
type GitStreamer struct {
        repo *gogit.Repository
}

// NewGitStreamer creates a new GitStreamer for the given repository
func NewGitStreamer(repo *gogit.Repository) *GitStreamer {
        return &GitStreamer{
                repo: repo,
        }
}

// PackfileStreamRequest represents a request for streaming a packfile
type PackfileStreamRequest struct {
        Wants        []string // Object hashes that the client wants
        Haves        []string // Object hashes that the client already has
        Capabilities []string // Client capabilities
}

// StreamPackfile streams a packfile to the provided writer using efficient streaming
// without loading all objects into memory at once
func (gs *GitStreamer) StreamPackfile(ctx context.Context, req *PackfileStreamRequest, w io.Writer) error {
        // Create a pipe for streaming data
        pipeReader, pipeWriter := io.Pipe()

        // Channel for error communication
        errChan := make(chan error, 1)

        // Start packfile generation in a goroutine
        go func() {
                defer func() { _ = pipeWriter.Close() }()
                err := gs.generatePackfileStream(ctx, req, pipeWriter)
                errChan <- err
        }()

        // Copy from pipe to destination writer (gateway response)
        _, copyErr := io.Copy(w, pipeReader)

        // Wait for generation to complete and get any errors
        generateErr := <-errChan

        // Return the first error encountered
        if generateErr != nil {
                return fmt.Errorf("packfile generation failed: %w", generateErr)
        }
        if copyErr != nil {
                return fmt.Errorf("packfile streaming failed: %w", copyErr)
        }

        return nil
}

// generatePackfileStream generates packfile data and streams it through the pipe
func (gs *GitStreamer) generatePackfileStream(ctx context.Context, req *PackfileStreamRequest, w io.Writer) error {
        // Create a packet line writer for the stream
        pktWriter := &StreamingPktLineWriter{w: w}

        // Send negotiation response
        if err := pktWriter.writeLine("NAK"); err != nil {
                return fmt.Errorf("failed to write NAK: %w", err)
        }

        // Send packfile start indicator
        if err := pktWriter.writeLine("0008NAK\n"); err != nil {
                return err
        }

        // Flush negotiation phase
        if err := pktWriter.flush(); err != nil {
                return err
        }

        // Stream objects without collecting them all in memory
        return gs.streamObjects(ctx, req, pktWriter)
}

// streamObjects streams Git objects one by one without accumulating in memory
func (gs *GitStreamer) streamObjects(ctx context.Context, req *PackfileStreamRequest, pktWriter *StreamingPktLineWriter) error {
        // Track processed objects to avoid duplicates
        processed := make(map[plumbing.Hash]bool)
        objectCount := 0

        // Send simplified packfile header
        if err := pktWriter.writeLine("# streaming pack\n"); err != nil {
                return err
        }

        // Stream each wanted object and its dependencies
        for _, want := range req.Wants {
                hash := plumbing.NewHash(want)

                // Check if client already has this object
                if gs.clientHasObject(hash.String(), req.Haves) {
                        continue
                }

                // Stream this object and its dependencies
                err := gs.streamObjectTree(ctx, hash, processed, req.Haves, pktWriter, &objectCount)
                if err != nil {
                        return fmt.Errorf("failed to stream object tree for %s: %w", want, err)
                }

                // Check for cancellation
                select {
                case <-ctx.Done():
                        return ctx.Err()
                default:
                        // Continue processing
                }
        }

        // Send packfile summary
        summary := fmt.Sprintf("# pack completed: %d objects streamed\n", objectCount)
        if err := pktWriter.writeLine(summary); err != nil {
                return err
        }

        // End with flush packet
        return pktWriter.flush()
}

// streamObjectTree streams an object and its dependencies without loading everything into memory
func (gs *GitStreamer) streamObjectTree(ctx context.Context, hash plumbing.Hash, processed map[plumbing.Hash]bool, haves []string, pktWriter *StreamingPktLineWriter, objectCount *int) error {
        // Skip if already processed or client has it
        if processed[hash] || gs.clientHasObject(hash.String(), haves) {
                return nil
        }

        // Mark as processed
        processed[hash] = true

        // Try to get the object
        obj, err := gs.repo.Object(plumbing.AnyObject, hash)
        if err != nil {
                // Object not found - skip silently (might be a partial clone scenario)
                return nil
        }

        // Stream this object
        objectInfo := fmt.Sprintf("object %s %s\n", hash.String(), obj.Type().String())
        if err := pktWriter.writeLine(objectInfo); err != nil {
                return err
        }

        *objectCount++

        // For commits, recursively stream parent commits
        if obj.Type() == plumbing.CommitObject {
                commit, err := object.GetCommit(gs.repo.Storer, hash)
                if err == nil {
                        // Stream parent commits (but limit depth to prevent memory issues)
                        for _, parentHash := range commit.ParentHashes {
                                // Recursive call for parent
                                err = gs.streamObjectTree(ctx, parentHash, processed, haves, pktWriter, objectCount)
                                if err != nil {
                                        return err
                                }

                                // Check for context cancellation
                                select {
                                case <-ctx.Done():
                                        return ctx.Err()
                                default:
                                        // Continue
                                }

                                // Safety limit to prevent infinite recursion
                                if *objectCount > 10000 {
                                        return fmt.Errorf("object count limit reached (10000) - stopping to prevent memory exhaustion")
                                }
                        }
                }
        }

        return nil
}

// clientHasObject checks if the client already has the specified object
func (gs *GitStreamer) clientHasObject(objectHash string, haves []string) bool {
        for _, have := range haves {
                if have == objectHash {
                        return true
                }
        }
        return false
}

// StreamingPktLineWriter provides packet line writing for streaming operations
type StreamingPktLineWriter struct {
        w  io.Writer
        mu sync.Mutex // Protect concurrent writes
}

// writeLine writes a Git protocol packet line
func (p *StreamingPktLineWriter) writeLine(data string) error {
        p.mu.Lock()
        defer p.mu.Unlock()

        // Calculate packet length (4 bytes for length + data)
        length := len(data) + 4
        line := fmt.Sprintf("%04x%s", length, data)

        _, err := p.w.Write([]byte(line))
        return err
}

// flush writes a flush packet (0000)
func (p *StreamingPktLineWriter) flush() error {
        p.mu.Lock()
        defer p.mu.Unlock()

        _, err := p.w.Write([]byte("0000"))
        return err
}

// StreamRefs streams repository references for info/refs endpoint
func (gs *GitStreamer) StreamRefs(ctx context.Context, w io.Writer, capabilities []string) error {
        pktWriter := &StreamingPktLineWriter{w: w}

        // Get all references from the repository
        refs, err := gs.repo.References()
        if err != nil {
                // Empty repository case - send dummy reference
                line := fmt.Sprintf("0000000000000000000000000000000000000000 capabilities^{}\x00%s\n",
                        fmt.Sprintf("%s agent=git/collabops-git-executor",
                                "multi_ack thin-pack side-band side-band-64k ofs-delta shallow no-progress include-tag multi_ack_detailed no-done symref=HEAD:refs/heads/main"))
                return pktWriter.writeLine(line)
        }

        firstRef := true
        refCount := 0

        // Stream each reference
        err = refs.ForEach(func(ref *plumbing.Reference) error {
                select {
                case <-ctx.Done():
                        return ctx.Err()
                default:
                }

                // Format reference line
                var line string
                if firstRef {
                        // First reference includes capabilities
                        capString := "multi_ack thin-pack side-band side-band-64k ofs-delta shallow no-progress include-tag multi_ack_detailed no-done symref=HEAD:refs/heads/main agent=git/collabops-git-executor"
                        line = fmt.Sprintf("%s %s\x00%s\n", ref.Hash().String(), ref.Name().String(), capString)
                        firstRef = false
                } else {
                        // Subsequent references without capabilities
                        line = fmt.Sprintf("%s %s\n", ref.Hash().String(), ref.Name().String())
                }

                refCount++
                return pktWriter.writeLine(line)
        })

        if err != nil {
                return fmt.Errorf("failed to stream references: %w", err)
        }

        // If no refs were streamed (empty repo), send dummy reference
        if refCount == 0 {
                dummyLine := fmt.Sprintf("0000000000000000000000000000000000000000 capabilities^{}\x00%s\n",
                        "multi_ack thin-pack side-band side-band-64k ofs-delta shallow no-progress include-tag multi_ack_detailed no-done symref=HEAD:refs/heads/main agent=git/collabops-git-executor")
                if err := pktWriter.writeLine(dummyLine); err != nil {
                        return err
                }
        }

        // End with flush packet
        return pktWriter.flush()
}
